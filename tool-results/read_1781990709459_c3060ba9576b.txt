     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef, useMemo } from "react";
     4→import { trackEvent } from "@/lib/analytics";
     5→import {
     6→  ChevronRight,
     7→  ChevronLeft,
     8→  ChevronDown,
     9→  Maximize2,
    10→  Minimize2,
    11→  CheckCircle2,
    12→  Lock,
    13→  Loader2,
    14→  BookOpen,
    15→  Trophy,
    16→  FileText,
    17→  AlertTriangle,
    18→  Rocket,
    19→} from "lucide-react";
    20→import ReactMarkdown from "react-markdown";
    21→import { useAppStore, type CourseData, type CourseChapter, type QuizQuestion } from "@/lib/store";
    22→import { t } from "@/lib/i18n";
    23→import Confetti from "@/components/coursia/Confetti";
    24→import CoursiaLogo from "@/components/coursia/CoursiaLogo";
    25→
    26→const LEVEL_EMOJIS = ["🌱", "⚡", "🔥"];
    27→const LEVEL_NAMES_FR = ["Débutant", "Intermédiaire", "Avancé"];
    28→const LEVEL_NAMES_EN = ["Beginner", "Intermediate", "Advanced"];
    29→
    30→export default function CourseViewer() {
    31→  const lang = useAppStore((s) => s.lang);
    32→  const tx = t(lang);
    33→  const user = useAppStore((s) => s.user);
    34→
    35→  const [course, setCourse] = useState<CourseData | null>(null);
    36→  const [loading, setLoading] = useState(true);
    37→  const [fetchError, setFetchError] = useState(false);
    38→  const [studySessionId, setStudySessionId] = useState<string | null>(null);
    39→  const [showConfetti, setShowConfetti] = useState(false);
    40→  const [courseCompleted, setCourseCompleted] = useState(false);
    41→  const [isCompleting, setIsCompleting] = useState(false);
    42→  const [mobileChapterOpen, setMobileChapterOpen] = useState(false);
    43→
    44→  // Subscription state for free chapter limit
    45→  const [isSubscribed, setIsSubscribed] = useState(false);
    46→  const [freeChapterLimit, setFreeChapterLimit] = useState(1);
    47→
    48→  // Level system states
    49→  const [isGeneratingLevel, setIsGeneratingLevel] = useState(false);
    50→  const [showReviewScreen, setShowReviewScreen] = useState(false);
    51→  const [completedLevel, setCompletedLevel] = useState(-1);
    52→  const [showStopConfirm, setShowStopConfirm] = useState(false);
    53→  const [showAllMastered, setShowAllMastered] = useState(false);
    54→
    55→  const selectedCourseId = useAppStore((s) => s.selectedCourseId);
    56→  const currentChapterIndex = useAppStore((s) => s.currentChapterIndex);
    57→  const setCurrentChapterIndex = useAppStore((s) => s.setCurrentChapterIndex);
    58→  const isFullscreen = useAppStore((s) => s.isFullscreen);
    59→  const setIsFullscreen = useAppStore((s) => s.setIsFullscreen);
    60→  const showFinalQuiz = useAppStore((s) => s.showFinalQuiz);
    61→  const setShowFinalQuiz = useAppStore((s) => s.setShowFinalQuiz);
    62→  const setView = useAppStore((s) => s.setView);
    63→  const showCelebration = useAppStore((s) => s.showCelebration);
    64→  const setShowCelebration = useAppStore((s) => s.setShowCelebration);
    65→  const setCelebrationMessage = useAppStore((s) => s.setCelebrationMessage);
    66→  const setRandomTopic = useAppStore((s) => s.setRandomTopic);
    67→  const setRandomCourseLang = useAppStore((s) => s.setRandomCourseLang);
    68→
    69→  const chapterListRef = useRef<HTMLDivElement>(null);
    70→  const celebrationTimerRef = useRef<ReturnType<typeof setTimeout>>();
    71→  const finalQuizTimerRef = useRef<ReturnType<typeof setTimeout>>();
    72→
    73→  // ── Expanded chapters state for sub-chapter display ──
    74→  const [expandedChapters, setExpandedChapters] = useState<Set<number>>(new Set());
    75→
    76→  const toggleChapterExpanded = (idx: number) => {
    77→    setExpandedChapters((prev) => {
    78→      const next = new Set(prev);
    79→      if (next.has(idx)) next.delete(idx); else next.add(idx);
    80→      return next;
    81→    });
    82→  };
    83→
    84→  // ── Parse sub-chapters from markdown content ──
    85→  const parseSubChapters = useCallback((content: string): string[] => {
    86→    return content.match(/^## (.+)$/gm)?.map((s) => s.replace(/^## /, "")) || [];
    87→  }, []);
    88→
    89→  const chapterSubChapters = useMemo(() => {
    90→    if (!course) return {} as Record<string, string[]>;
    91→    const map: Record<string, string[]> = {};
    92→    for (const ch of course.chapters) map[ch.id] = parseSubChapters(ch.content || "");
    93→    return map;
    94→  }, [course, parseSubChapters]);
    95→
    96→  // ── Group chapters by level ──
    97→  const chaptersByLevel = useMemo(() => {
    98→    if (!course) return [];
    99→    const levels: { level: number; chapters: { chapter: CourseChapter; globalIdx: number }[] }[] = [];
   100→    let currentLevel = -1;
   101→    for (let i = 0; i < course.chapters.length; i++) {
   102→      const ch = course.chapters[i];
   103→      const chLevel = ch.level ?? 0;
   104→      if (chLevel !== currentLevel) {
   105→        levels.push({ level: chLevel, chapters: [] });
   106→        currentLevel = chLevel;
   107→      }
   108→      levels[levels.length - 1].chapters.push({ chapter: ch, globalIdx: i });
   109→    }
   110→    return levels;
   111→  }, [course]);
   112→
   113→  // ── Get current max level and stopped state ──
   114→  const maxUnlockedLevel = course?.maxUnlockedLevel ?? 0;
   115→  const stoppedAtLevel = course?.stoppedAtLevel ?? -1;
   116→  const isStopped = stoppedAtLevel >= 0;
   117→
   118→  // ── Completed count and progress ──
   119→  const completedCount = useMemo(() => {
   120→    if (!course) return 0;
   121→    return course.chapters.filter((ch) => ch.progress?.completed).length;
   122→  }, [course]);
   123→
   124→  const totalChapters = course?.chapters.length ?? 0;
   125→  const overallProgress = totalChapters > 0 ? Math.round((completedCount / totalChapters) * 100) : 0;
   126→
   127→  // Check if all chapters in current max level are completed
   128→  const allCurrentLevelCompleted = useMemo(() => {
   129→    if (!course) return false;
   130→    const levelChapters = course.chapters.filter((ch) => (ch.level ?? 0) <= maxUnlockedLevel);
   131→    if (levelChapters.length === 0) return false;
   132→    return levelChapters.every((ch) => ch.progress?.completed);
   133→  }, [course, maxUnlockedLevel]);
   134→
   135→  const allChaptersCompleted = totalChapters > 0 && completedCount === totalChapters;
   136→
   137→  // ── Check if a chapter is level-locked ──
   138→  const isChapterLevelLocked = (index: number): boolean => {
   139→    if (!course) return false;
   140→    const ch = course.chapters[index];
   141→    const chLevel = ch.level ?? 0;
   142→    // Locked if chapter level exceeds maxUnlockedLevel
   143→    if (chLevel > maxUnlockedLevel) return true;
   144→    // If stopped, lock chapters above stoppedAtLevel
   145→    if (isStopped && chLevel > stoppedAtLevel) return true;
   146→    return false;
   147→  };
   148→
   149→  const isChapterUnlocked = (index: number) => {
   150→    if (isChapterLevelLocked(index)) return false;
   151→    // Free preview: lock chapters beyond freeChapterLimit for non-subscribers
   152→    if (!isSubscribed && index >= freeChapterLimit) return false;
   153→    if (index === 0) return true;
   154→    return course?.chapters[index - 1]?.progress?.completed === true;
   155→  };
   156→
   157→  // ── Study session tracking ──
   158→  const startStudySession = useCallback(async (cId: string, chId?: string) => {
   159→    try {
   160→      const res = await fetch("/api/study-time", {
   161→        method: "POST",
   162→        headers: { "Content-Type": "application/json" },
   163→        body: JSON.stringify({ action: "start", courseId: cId, chapterId: chId }),
   164→      });
   165→      if (res.ok) {
   166→        const data = await res.json();
   167→        setStudySessionId(data.sessionId);
   168→      }
   169→    } catch { /* ignore */ }
   170→  }, []);
   171→
   172→  const endStudySession = useCallback(async () => {
   173→    if (!studySessionId) return;
   174→    try {
   175→      await fetch("/api/study-time", {
   176→        method: "POST",
   177→        headers: { "Content-Type": "application/json" },
   178→        body: JSON.stringify({ action: "end", sessionId: studySessionId }),
   179→      });
   180→    } catch { /* ignore */ }
   181→    setStudySessionId(null);
   182→  }, [studySessionId]);
   183→
   184→  useEffect(() => {
   185→    return () => {
   186→      if (celebrationTimerRef.current) clearTimeout(celebrationTimerRef.current);
   187→      if (finalQuizTimerRef.current) clearTimeout(finalQuizTimerRef.current);
   188→    };
   189→  }, []);
   190→
   191→  useEffect(() => {
   192→    return () => {
   193→      if (studySessionId) {
   194→        fetch("/api/study-time", {
   195→          method: "POST",
   196→          headers: { "Content-Type": "application/json" },
   197→          body: JSON.stringify({ action: "end", sessionId: studySessionId }),
   198→        }).catch(() => {});
   199→      }
   200→    };
   201→  }, [studySessionId]);
   202→
   203→  useEffect(() => {
   204→    const handleBeforeUnload = () => {
   205→      if (studySessionId) {
   206→        navigator.sendBeacon("/api/study-time", new Blob([JSON.stringify({ action: "end", sessionId: studySessionId })], { type: "application/json" }));
   207→      }
   208→    };
   209→    window.addEventListener("beforeunload", handleBeforeUnload);
   210→    return () => window.removeEventListener("beforeunload", handleBeforeUnload);
   211→  }, [studySessionId]);
   212→
   213→  const fetchCourse = useCallback(async () => {
   214→    if (!selectedCourseId) return;
   215→    setLoading(true);
   216→    setFetchError(false);
   217→    try {
   218→      // Fetch subscription status in parallel with course data
   219→      const userId = useAppStore.getState().userId;
   220→      const authHeaders: Record<string, string> = {};
   221→      if (userId) authHeaders["Authorization"] = `Bearer ${userId}`;
   222→
   223→      const [courseRes, statusRes] = await Promise.all([
   224→        fetch(`/api/courses/${selectedCourseId}`),
   225→        userId ? fetch("/api/courses/paywall-status", { headers: authHeaders }) : Promise.resolve(null),
   226→      ]);
   227→
   228→      const data = await courseRes.json();
   229→      // Parse subscription status
   230→      if (statusRes && statusRes.ok) {
   231→        const pw = await statusRes.json();
   232→        setIsSubscribed(!!pw.hasSubscription && pw.subscriptionStatus === "active");
   233→        setFreeChapterLimit(pw.freeChapterLimit || 1);
   234→      }
   235→
   236→      if (courseRes.ok && data.chapters?.length > 0) {
   237→        setCourse(data);
   238→        if (data.courseCompleted) setCourseCompleted(true);
   239→        const savedChapterKey = `coursia-last-chapter-${selectedCourseId}`;
   240→        const savedChapter = typeof window !== "undefined" ? localStorage.getItem(savedChapterKey) : null;
   241→        let restoreIdx = 0;
   242→        if (savedChapter) {
   243→          const savedIdx = parseInt(savedChapter, 10);
   244→          if (!isNaN(savedIdx) && savedIdx >= 0 && savedIdx < data.chapters.length) {
   245→            if (!isChapterLevelLockedForData(savedIdx, data)) restoreIdx = savedIdx;
   246→          }
   247→        }
   248→        if (restoreIdx === 0) {
   249→          const firstIncomplete = data.chapters.findIndex((ch: CourseChapter) => !ch.progress?.completed && !isChapterLevelLockedForData(data.chapters.indexOf(ch), data));
   250→          if (firstIncomplete >= 0) restoreIdx = firstIncomplete;
   251→        }
   252→        const currentIdx = useAppStore.getState().currentChapterIndex;
   253→        if (currentIdx === 0 || currentIdx >= data.chapters.length) {
   254→          setCurrentChapterIndex(restoreIdx);
   255→        }
   256→        startStudySession(selectedCourseId, data.chapters[restoreIdx >= 0 ? restoreIdx : 0]?.id);
   257→      } else {
   258→        setFetchError(true);
   259→      }
   260→    } catch {
   261→      setFetchError(true);
   262→    } finally {
   263→      setLoading(false);
   264→    }
   265→  }, [selectedCourseId, setCurrentChapterIndex]);
   266→
   267→  // Helper to check level lock for data during fetch (before state is set)
   268→  const isChapterLevelLockedForData = (index: number, data: CourseData): boolean => {
   269→    const ch = data.chapters[index];
   270→    const chLevel = ch.level ?? 0;
   271→    if (chLevel > (data.maxUnlockedLevel ?? 0)) return true;
   272→    if ((data.stoppedAtLevel ?? -1) >= 0 && chLevel > (data.stoppedAtLevel ?? -1)) return true;
   273→    return false;
   274→  };
   275→
   276→  useEffect(() => { fetchCourse(); }, [selectedCourseId, fetchCourse]);
   277→
   278→  useEffect(() => {
   279→    if (!chapterListRef.current) return;
   280→    const activeEl = chapterListRef.current.querySelector(`[data-chapter-idx="${currentChapterIndex}"]`);
   281→    if (activeEl) activeEl.scrollIntoView({ behavior: "smooth", block: "nearest" });
   282→  }, [currentChapterIndex]);
   283→
   284→  useEffect(() => {
   285→    if (!course) return;
   286→    const activeCh = course.chapters[currentChapterIndex];
   287→    if (!activeCh) return;
   288→    const subs = chapterSubChapters[activeCh.id];
   289→    if (subs && subs.length > 0) {
   290→      setExpandedChapters((prev) => {
   291→        if (prev.has(currentChapterIndex)) return prev;
   292→        const next = new Set(prev);
   293→        next.add(currentChapterIndex);
   294→        return next;
   295→      });
   296→    }
   297→  }, [currentChapterIndex, course, chapterSubChapters]);
   298→
   299→  useEffect(() => {
   300→    if (!selectedCourseId || !course) return;
   301→    localStorage.setItem(`coursia-last-chapter-${selectedCourseId}`, String(currentChapterIndex));
   302→  }, [currentChapterIndex, selectedCourseId, course]);
   303→
   304→  const currentChapter = course?.chapters[currentChapterIndex];
   305→
   306→  const completeCurrentChapter = useCallback(async (): Promise<boolean> => {
   307→    if (!currentChapter || currentChapter.progress?.completed) return true;
   308→    try {
   309→      const res = await fetch(`/api/courses/${selectedCourseId}/chapters/${currentChapter.id}/complete`, {
   310→        method: "POST",
   311→        headers: { "Content-Type": "application/json" },
   312→      });
   313→      if (res.ok) {
   314→        const courseRes = await fetch(`/api/courses/${selectedCourseId}`);
   315→        if (courseRes.ok) setCourse(await courseRes.json());
   316→        return true;
   317→      }
   318→      return false;
   319→    } catch { return false; }
   320→  }, [currentChapter, selectedCourseId]);
   321→
   322→  // ── Paywall redirect: will clicking "next" send user to offers? ──
   323→  const isPaywallRedirect = useMemo(() => {
   324→    if (isSubscribed) return false;
   325→    if (!course) return false;
   326→    const nextIdx = currentChapterIndex + 1;
   327→    if (nextIdx >= course.chapters.length) return false;
   328→    return nextIdx >= freeChapterLimit;
   329→  }, [isSubscribed, course, currentChapterIndex, freeChapterLimit]);
   330→
   331→  const goToNext = useCallback(async () => {
   332→    if (!course || isCompleting) return;
   333→    if (currentChapterIndex >= course.chapters.length - 1) return;
   334→
   335→    // Free preview: block chapter 2+ for non-subscribers
   336→    if (!isSubscribed && currentChapterIndex + 1 >= freeChapterLimit) {
   337→      trackEvent({ name: "paywall_hit" });
   338→      setView("offers");
   339→      return;
   340→    }
   341→
   342→    if (isChapterLevelLocked(currentChapterIndex + 1)) return;
   343→
   344→    setIsCompleting(true);
   345→    const wasJustCompleted = !currentChapter?.progress?.completed;
   346→    if (wasJustCompleted) {
   347→      const success = await completeCurrentChapter();
   348→      if (!success) { setIsCompleting(false); return; }
   349→      const userName = user?.firstName || (lang === "fr" ? "Champion" : "Champion");
   350→      setShowConfetti(true);
   351→      setShowCelebration(true);
   352→      setCelebrationMessage(lang === "fr" ? `Chapitre ${currentChapterIndex + 1} terminé ${userName} ! 🎉` : `Chapter ${currentChapterIndex + 1} done ${userName}! 🎉`);
   353→      if (celebrationTimerRef.current) clearTimeout(celebrationTimerRef.current);
   354→      celebrationTimerRef.current = setTimeout(() => { setShowCelebration(false); setShowConfetti(false); }, 2000);
   355→    }
   356→    endStudySession();
   357→    const nextIdx = currentChapterIndex + 1;
   358→    setCurrentChapterIndex(nextIdx);
   359→    startStudySession(course.id, course.chapters[nextIdx]?.id);
   360→    setIsCompleting(false);
   361→  }, [course, isCompleting, currentChapter?.progress?.completed, currentChapterIndex, completeCurrentChapter, user?.firstName, lang, endStudySession, startStudySession, setCurrentChapterIndex, isSubscribed, freeChapterLimit, setView]);
   362→
   363→  const goToPrev = useCallback(() => {
   364→    if (currentChapterIndex === 0 || !course) return;
   365→    endStudySession();
   366→    const prevIdx = currentChapterIndex - 1;
   367→    setCurrentChapterIndex(prevIdx);
   368→    startStudySession(course.id, course.chapters[prevIdx]?.id);
   369→  }, [currentChapterIndex, course, endStudySession, startStudySession, setCurrentChapterIndex]);
   370→
   371→  // ── Level Complete handler ──
   372→  const handleFinalQuizComplete = useCallback((passed: boolean) => {
   373→    if (!passed) return;
   374→    const userName = user?.firstName || (lang === "fr" ? "Champion" : "Champion");
   375→    setShowConfetti(true);
   376→    setShowCelebration(true);
   377→    setCelebrationMessage(lang === "fr" ? `Félicitations ${userName} ! 🏆` : `Congratulations ${userName}! 🏆`);
   378→    setCourseCompleted(true);
   379→    if (finalQuizTimerRef.current) clearTimeout(finalQuizTimerRef.current);
   380→    finalQuizTimerRef.current = setTimeout(() => {
   381→      setShowCelebration(false);
   382→      setShowFinalQuiz(false);
   383→      setShowConfetti(false);
   384→      endStudySession();
   385→
   386→      const courseLevel = course?.level ?? 0;
   387→      const currentMaxLevel = course?.maxUnlockedLevel ?? 0;
   388→      setCompletedLevel(currentMaxLevel);
   389→
   390→      if (courseLevel >= 2 && currentMaxLevel >= 2) {
   391→        // All levels mastered!
   392→        setShowAllMastered(true);
   393→      } else {
   394→        // Show level-up modal
   395→        setShowReviewScreen(true);
   396→      }
   397→      fetchCourse();
   398→    }, 4000);
   399→  }, [course, user?.firstName, lang, endStudySession, fetchCourse]);
   400→
   401→  // ── Continue to next level ──
   402→  const handleContinueToNextLevel = useCallback(async () => {
   403→    if (!course || !selectedCourseId) return;
   404→    const nextLevel = maxUnlockedLevel + 1;
   405→    if (nextLevel > 2) return;
   406→
   407→    setIsGeneratingLevel(true);
   408→    setShowReviewScreen(false);
   409→    setShowConfetti(false);
   410→
   411→    try {
   412→      const res = await fetch(`/api/courses/${selectedCourseId}/generate-level`, {
   413→        method: "POST",
   414→        headers: { "Content-Type": "application/json" },
   415→        body: JSON.stringify({ level: nextLevel }),
   416→      });
   417→      const data = await res.json();
   418→      if (res.ok && data.chapters?.length > 0) {
   419→        // Refetch course
   420→        await fetchCourse();
   421→        // Navigate to first chapter of new level
   422→        const firstNewChapterIdx = course.chapters.findIndex((ch) => (ch.level ?? 0) === nextLevel);
   423→        if (firstNewChapterIdx >= 0) setCurrentChapterIndex(firstNewChapterIdx);
   424→      }
   425→    } catch {
   426→      // show error
   427→    } finally {
   428→      setIsGeneratingLevel(false);
   429→    }
   430→  }, [course, selectedCourseId, maxUnlockedLevel, fetchCourse, setCurrentChapterIndex]);
   431→
   432→  // ── Stop at current level ──
   433→  const handleStopHere = useCallback(async () => {
   434→    if (!selectedCourseId) return;
   435→    try {
   436→      await fetch(`/api/courses/${selectedCourseId}/stop-level`, {
   437→        method: "POST",
   438→        headers: { "Content-Type": "application/json" },
   439→      });
   440→      setShowReviewScreen(false);
   441→      setShowConfetti(false);
   442→      setShowStopConfirm(true);
   443→      fetchCourse();
   444→    } catch {
   445→      // ignore
   446→    }
   447→  }, [selectedCourseId, fetchCourse]);
   448→
   449→  const handleCloseStopConfirm = () => {
   450→    setShowStopConfirm(false);
   451→  };
   452→
   453→  const handleCloseAllMastered = () => {
   454→    setShowAllMastered(false);
   455→    setView("library");
   456→  };
   457→
   458→  // Keyboard navigation
   459→  useEffect(() => {
   460→    const handleKey = (e: KeyboardEvent) => {
   461→      if (isFullscreen) {
   462→        if (e.key === "ArrowRight" || e.key === " ") { e.preventDefault(); goToNext(); }
   463→        if (e.key === "ArrowLeft") { e.preventDefault(); goToPrev(); }
   464→        if (e.key === "Escape") setIsFullscreen(false);
   465→      }
   466→    };
   467→    window.addEventListener("keydown", handleKey);
   468→    return () => window.removeEventListener("keydown", handleKey);
   469→  }, [isFullscreen, goToNext, goToPrev, setIsFullscreen]);
   470→
   471→  // ── Level names helper ──
   472→  const getLevelName = (level: number) => lang === "fr" ? LEVEL_NAMES_FR[level] : LEVEL_NAMES_EN[level];
   473→
   474→  // ══════════════════════════════════════════════════════════════════
   475→  // Loading / error state
   476→  // ══════════════════════════════════════════════════════════════════
   477→  if (!selectedCourseId || fetchError || (!loading && !course)) {
   478→    return (
   479→      <div className="flex items-center justify-center min-h-screen animate-fade-in">
   480→        <div className="text-center">
   481→          <div className="text-5xl mb-4 opacity-40">📚</div>
   482→          <p className="text-muted-foreground text-lg mb-6">{tx.viewer.back}</p>
   483→          <button onClick={() => setView("landing")} className="px-6 py-3 rounded-full bg-gradient-to-r from-mauve to-mauve-dark text-white font-bold hover:from-mauve-light hover:to-mauve transition-all cursor-pointer">{tx.viewer.back}</button>
   484→        </div>
   485→      </div>
   486→    );
   487→  }
   488→
   489→  if (loading || !course || !currentChapter) {
   490→    return (
   491→      <div className="flex items-center justify-center min-h-screen animate-fade-in">
   492→        <div className="text-center">
   493→          <Loader2 className="w-10 h-10 animate-spin text-mauve mx-auto mb-4" />
   494→          <p className="text-muted-foreground text-lg">{tx.viewer.loading}</p>
   495→        </div>
   496→      </div>
   497→    );
   498→  }
   499→
   500→  // ══════════════════════════════════════════════════════════════════
   501→  // GENERATING NEXT LEVEL — full screen loading
   502→  // ══════════════════════════════════════════════════════════════════
   503→  if (isGeneratingLevel) {
   504→    return (
   505→      <>

  1103→  const tx = t(lang);
  1104→
