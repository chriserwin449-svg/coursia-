     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef, useMemo } from "react";
     4→import {
     5→  Sparkles,
     6→  Plus,
     7→  X,
     8→  Link as LinkIcon,
     9→  Loader2,
    10→  BookOpen,
    11→  ChevronRight,
    12→  Globe,
    13→  Crown,
    14→  CreditCard,
    15→} from "lucide-react";
    16→import { useAppStore } from "@/lib/store";
    17→import { t } from "@/lib/i18n";
    18→import type { CourseData } from "@/lib/store";
    19→
    20→export default function CreateCourse() {
    21→  const lang = useAppStore((s) => s.lang);
    22→  const tx = t(lang);
    23→  const user = useAppStore((s) => s.user);
    24→
    25→  const setView = useAppStore((s) => s.setView);
    26→  const setSelectedCourseId = useAppStore((s) => s.setSelectedCourseId);
    27→  const setIsGenerating = useAppStore((s) => s.setIsGenerating);
    28→
    29→  const [title, setTitle] = useState("");
    30→  const [links, setLinks] = useState<string[]>([]);
    31→  const [linkInput, setLinkInput] = useState("");
    32→  const [loading, setLoading] = useState(false);
    33→  const [error, setError] = useState("");
    34→  const [courseLang, setCourseLang] = useState("fr"); // "fr" or "en"
    35→  const [placeholderIndex, setPlaceholderIndex] = useState(0);
    36→  const [courses, setCourses] = useState<CourseData[]>([]);
    37→  const [showSuggested, setShowSuggested] = useState(false);
    38→  const [suggestedTopic, setSuggestedTopic] = useState("");
    39→  const [hasSubscription, setHasSubscription] = useState(false);
    40→  const [coursesRemaining, setCoursesRemaining] = useState(3);
    41→  const [inTrial, setInTrial] = useState(false);
    42→  const [trialExpired, setTrialExpired] = useState(false);
    43→  const [inGracePeriod, setInGracePeriod] = useState(false);
    44→  const [trialDaysRemaining, setTrialDaysRemaining] = useState(0);
    45→  const [trialCoursesMax, setTrialCoursesMax] = useState(3);
    46→  const [selectedLevel, setSelectedLevel] = useState(0);
    47→  const [isRandomTopic, setIsRandomTopic] = useState(false);
    48→  const [requireCard, setRequireCard] = useState(false);
    49→  const [showCardModal, setShowCardModal] = useState(false);
    50→  const [cardVerifying, setCardVerifying] = useState(false);
    51→
    52→  // ─── Store refs for random topic ────────────────────────────────────────
    53→  const storeRandomTopic = useAppStore((s) => s.randomTopic);
    54→  const storeRandomCourseLang = useAppStore((s) => s.randomCourseLang);
    55→  const setStoreRandomTopic = useAppStore((s) => s.setRandomTopic);
    56→  const prevRandomRef = useRef<string | null>(null);
    57→
    58→  // ─── React to random topic changes from TopBar (instant, no reload) ───
    59→  useEffect(() => {
    60→    if (storeRandomTopic && storeRandomTopic !== prevRandomRef.current) {
    61→      setTitle(storeRandomTopic);
    62→      setSuggestedTopic(storeRandomTopic);
    63→      setShowSuggested(true);
    64→      setIsRandomTopic(true);
    65→      setSelectedLevel(0);
    66→      if (storeRandomCourseLang === "fr" || storeRandomCourseLang === "en") {
    67→        setCourseLang(storeRandomCourseLang);
    68→      }
    69→      prevRandomRef.current = storeRandomTopic;
    70→      setStoreRandomTopic(null); // consume it
    71→    }
    72→  }, [storeRandomTopic, storeRandomCourseLang, setStoreRandomTopic]);
    73→
    74→  // ─── Personalized greeting (time-based) ──────────────────────────────
    75→  const greeting = useMemo(() => {
    76→    const hour = new Date().getHours();
    77→    const firstName = user?.firstName || "";
    78→    const casualGreetings = tx.create.greetingCasual;
    79→    const messages = tx.create.greetingMessages;
    80→
    81→    // Pick a greeting based on time of day
    82→    let timeGreeting: string;
    83→    if (hour >= 5 && hour < 12) {
    84→      timeGreeting = tx.create.greetingMorning;
    85→    } else if (hour >= 12 && hour < 17) {
    86→      timeGreeting = tx.create.greetingAfternoon;
    87→    } else if (hour >= 17 && hour < 22) {
    88→      timeGreeting = tx.create.greetingEvening;
    89→    } else {
    90→      // Late night - use a random casual greeting
    91→      timeGreeting = casualGreetings[Math.floor(Math.random() * casualGreetings.length)];
    92→    }
    93→
    94→    // Pick a random encouraging message (seeded by day to avoid changing on every render)
    95→    const dayIndex = Math.floor(Date.now() / 86_400_000); // changes once per day
    96→    const message = messages[dayIndex % messages.length];
    97→
    98→    return {
    99→      greeting: timeGreeting,
   100→      name: firstName,
   101→      message,
   102→    };
   103→  }, [user?.firstName, tx.create]);
   104→
   105→  // ─── Rotating placeholder with typing/fade effect ────────────────────
   106→  const placeholders = tx.create.placeholders;
   107→  const [displayedPlaceholder, setDisplayedPlaceholder] = useState(placeholders[0]);
   108→  const [isTyping, setIsTyping] = useState(true);
   109→  const [charIndex, setCharIndex] = useState(0);
   110→  const [isFading, setIsFading] = useState(false);
   111→  const innerTimerRef = useRef<ReturnType<typeof setTimeout>>();
   112→
   113→  useEffect(() => {
   114→    // Skip typing animation while user has typed something
   115→    if (title.length > 0) {
   116→      setDisplayedPlaceholder("");
   117→      setIsTyping(false);
   118→      return;
   119→    }
   120→
   121→    const currentPlaceholder = placeholders[placeholderIndex];
   122→
   123→    if (isTyping) {
   124→      if (charIndex <= currentPlaceholder.length) {
   125→        const timeout = setTimeout(() => {
   126→          setDisplayedPlaceholder(currentPlaceholder.slice(0, charIndex));
   127→          setCharIndex(charIndex + 1);
   128→        }, 35);
   129→        return () => clearTimeout(timeout);
   130→      } else {
   131→        // Finished typing, wait then start fading
   132→        const timeout = setTimeout(() => {
   133→          setIsFading(true);
   134→          innerTimerRef.current = setTimeout(() => {
   135→            setPlaceholderIndex((placeholderIndex + 1) % placeholders.length);
   136→            setCharIndex(0);
   137→            setIsTyping(true);
   138→            setIsFading(false);
   139→          }, 500);
   140→        }, 2200);
   141→        return () => { clearTimeout(timeout); if (innerTimerRef.current) clearTimeout(innerTimerRef.current); };
   142→      }
   143→    }
   144→  }, [charIndex, isTyping, isFading, placeholderIndex, placeholders, title]);
   145→
   146→  // ─── Fetch courses & subscription status ───────────────────────────
   147→  const fetchCourses = useCallback(async () => {
   148→    try {
   149→      const res = await fetch(`/api/courses?userId=${useAppStore.getState().userId || ''}`);
   150→      if (res.ok) {
   151→        const data = await res.json();
   152→        const list = (data.courses as CourseData[]) || [];
   153→        setCourses(list);
   154→      }
   155→
   156→      // Check subscription & trial status via paywall-status API
   157→      const userId = useAppStore.getState().userId;
   158→      const headers: Record<string, string> = {};
   159→      if (userId) headers["Authorization"] = `Bearer ${userId}`;
   160→      const pwRes = await fetch("/api/courses/paywall-status", { headers });
   161→      if (pwRes.ok) {
   162→        const pw = await pwRes.json();
   163→        setHasSubscription(pw.hasSubscription);
   164→        setInTrial(pw.inTrial);
   165→        setTrialExpired(pw.showPaywall && pw.paywallReason === "trial_expired");
   166→        setInGracePeriod(pw.inGracePeriod);
   167→        if (pw.inTrial) {
   168→          setTrialDaysRemaining(pw.trialDaysRemaining || 0);
   169→          setTrialCoursesMax(pw.trialCoursesMax || 3);
   170→          setCoursesRemaining(Math.max(0, (pw.trialCoursesMax || 3) - (pw.trialCoursesGenerated || 0)));
   171→          setRequireCard(!!pw.requireCard);
   172→        } else if (!pw.hasSubscription) {
   173→          setCoursesRemaining(Math.max(0, (pw.trialCoursesMax || 3) - (pw.trialCoursesGenerated || 0)));
   174→          setRequireCard(!!pw.requireCard);
   175→        } else {
   176→          setCoursesRemaining(999);
   177→          setRequireCard(false);
   178→        }
   179→      }
   180→    } catch {
   181→      // silently ignore
   182→    }
   183→  }, []);
   184→
   185→  useEffect(() => {
   186→    fetchCourses();
   187→  }, [fetchCourses]);
   188→
   189→  // ─── Clear suggested topic when user modifies title manually ────────
   190→  const handleTitleChange = (newTitle: string) => {
   191→    setTitle(newTitle);
   192→    if (newTitle.trim() !== suggestedTopic.trim()) {
   193→      setShowSuggested(false);
   194→      setSuggestedTopic("");
   195→      setIsRandomTopic(false);
   196→    }
   197→  };
   198→
   199→  // ─── Link management ──────────────────────────────────────────────────
   200→  const addLink = () => {
   201→    const trimmed = linkInput.trim();
   202→    if (trimmed && !links.includes(trimmed)) {
   203→      setLinks([...links, trimmed]);
   204→      setLinkInput("");
   205→    }
   206→  };
   207→
   208→  const removeLink = (index: number) => {
   209→    setLinks(links.filter((_, i) => i !== index));
   210→  };
   211→
   212→  // ─── Generate course ──────────────────────────────────────────────────
   213→  const generateCourse = async () => {
   214→    if (!title.trim()) return;
   215→
   216→    // Show card modal if required
   217→    if (requireCard && !hasSubscription) {
   218→      setShowCardModal(true);
   219→      return;
   220→    }
   221→
   222→    // Trial limit check
   223→    if (!hasSubscription && coursesRemaining <= 0) {
   224→      setView("offers");
   225→      return;
   226→    }
   227→
   228→    setLoading(true);
   229→    setError("");
   230→    setIsGenerating(true);
   231→
   232→    try {
   233→      const res = await fetch("/api/courses/generate", {
   234→        method: "POST",
   235→        headers: { "Content-Type": "application/json" },
   236→        body: JSON.stringify({
   237→          title: title.trim(),
   238→          sourceLinks: links,
   239→          level: isRandomTopic ? 0 : selectedLevel,
   240→          courseLang,
   241→          userId: useAppStore.getState().userId,
   242→        }),
   243→      });
   244→
   245→      const data = await res.json();
   246→      if (!res.ok) {
   247→        if (data.error === "FREE_LIMIT" || data.error === "TRIAL_LIMIT" || data.error === "TRIAL_EXPIRED") {
   248→          setView("offers");
   249→          return;
   250→        }
   251→        if (data.error === "REQUIRE_CARD") {
   252→          setShowCardModal(true);
   253→          return;
   254→        }
   255→        throw new Error(data.error || "Failed to generate");
   256→      }
   257→
   258→      const course = data.course as CourseData;
   259→      setSelectedCourseId(course.id);
   260→      setView("viewer");
   261→    } catch (err: unknown) {
   262→      setError(err instanceof Error ? err.message : "Une erreur est survenue");
   263→    } finally {
   264→      setLoading(false);
   265→      setIsGenerating(false);
   266→      fetchCourses();
   267→    }
   268→  };
   269→
   270→  // ─── Open a course ────────────────────────────────────────────────────
   271→  const openCourse = (id: string) => {
   272→    setSelectedCourseId(id);
   273→    setView("viewer");
   274→  };
   275→
   276→  // ─── Personalized trial counter text ────────────────────────────────
   277→  const trialCounterText = useMemo(() => {
   278→    const firstName = user?.firstName || "";
   279→
   280→    // Pre-trial (no courses yet, show simple welcome)
   281→    if (!inTrial && coursesRemaining === trialCoursesMax) {
   282→      if (firstName) {
   283→        return (tx.create.trialNoCourses as string).replace("{name}", firstName);
   284→      }
   285→      return lang === "fr"
   286→        ? `Tu as ${coursesRemaining} cours gratuit${coursesRemaining > 1 ? "s" : ""} sur Coursia !`
   287→        : `You have ${coursesRemaining} free course${coursesRemaining > 1 ? "s" : ""} on Coursia!`;
   288→    }
   289→
   290→    // Active trial — show detailed counter
   291→    if (inTrial) {
   292→      const dayWord = trialDaysRemaining > 1
   293→        ? (lang === "fr" ? tx.create.trialCounterDays : tx.create.trialCounterDays)
   294→        : (lang === "fr" ? tx.create.trialCounterDay : tx.create.trialCounterDay);
   295→      const courseSuffix = coursesRemaining > 1 ? "s" : "";
   296→      const daySuffix = trialDaysRemaining > 1 ? "s" : "";
   297→
   298→      if (firstName) {
   299→        return lang === "fr"
   300→          ? `${firstName}, il te reste ${coursesRemaining} cours gratuit${courseSuffix} sur ton temps d'essai · ${trialDaysRemaining} ${dayWord} restant${daySuffix}`
   301→          : `${firstName}, you have ${coursesRemaining} free course${courseSuffix} left on your trial · ${trialDaysRemaining} trial ${dayWord} remaining`;
   302→      }
   303→      return lang === "fr"
   304→        ? `Il te reste ${coursesRemaining} cours gratuit${courseSuffix} · ${trialDaysRemaining} ${dayWord} d'essai`
   305→        : `${coursesRemaining} free course${courseSuffix} · ${trialDaysRemaining} trial ${dayWord} remaining`;
   306→    }
   307→
   308→    // Post-trial, courses used up
   309→    return lang === "fr"
   310→      ? `${firstName ? firstName + ", " : ""}tu as utilisé tes ${trialCoursesMax} cours d'essai`
   311→      : `${firstName ? firstName + ", " : ""}you've used all ${trialCoursesMax} trial courses`;
   312→  }, [user?.firstName, lang, inTrial, coursesRemaining, trialDaysRemaining, trialCoursesMax, tx.create]);
   313→
   314→  // ─── Helpers ──────────────────────────────────────────────────────────
   315→  const courseLangLabels = tx.create.courseLangs;
   316→
   317→  const formatDate = (dateStr: string) => {
   318→    const d = new Date(dateStr);
   319→    return d.toLocaleDateString(lang === "fr" ? "fr-FR" : "en-US", {
   320→      day: "numeric",
   321→      month: "short",
   322→      year: "numeric",
   323→    });
   324→  };
   325→
   326→  return (
   327→    <>
   328→    <div className="max-w-3xl mx-auto px-4 sm:px-6 md:px-10 pt-14 sm:pt-20 pb-8 md:pt-24 md:pb-16">
   329→      {/* ═══════════ Personalized Greeting ═══════════ */}
   330→      {user && (
   331→        <div className="mb-6 sm:mb-8">
   332→          <div className="glass rounded-2xl sm:rounded-3xl p-5 sm:p-6 md:p-8 text-center">
   333→            <div className="text-4xl sm:text-5xl mb-3 animate-bounce" role="img" aria-label="waving hand">👋</div>
   334→            <h1 className="text-2xl sm:text-3xl md:text-4xl font-extrabold mb-2">
   335→              <span className="gradient-text">{greeting.greeting} {greeting.name} !</span>
   336→            </h1>
   337→            <p className="text-muted-foreground text-sm sm:text-base md:text-lg">
   338→              {greeting.message}
   339→            </p>
   340→          </div>
   341→        </div>
   342→      )}
   343→
   344→      {/* ═══════════ Main creation card ═══════════ */}
   345→      <div className="glass rounded-2xl sm:rounded-3xl p-4 sm:p-6 md:p-10">
   346→        {/* ─── Title input ─── */}
   347→        <div className="mb-8">
   348→          <label className="block text-sm font-bold mb-3 text-muted-foreground uppercase tracking-wider">
   349→            {tx.create.placeholder}
   350→          </label>
   351→          <div className="relative">
   352→            <input
   353→              type="text"
   354→              value={title}
   355→              onChange={(e) => handleTitleChange(e.target.value)}
   356→              placeholder=""
   357→              className="w-full px-6 py-5 rounded-2xl bg-night border border-border text-foreground text-lg font-bold placeholder:text-muted-foreground/50 focus:outline-none focus:border-mauve focus:ring-2 focus:ring-mauve/20 transition-all duration-300"
   358→              onKeyDown={(e) => e.key === "Enter" && generateCourse()}
   359→              disabled={loading}
   360→            />
   361→            {/* Animated placeholder overlay */}
   362→            {!title && (
   363→              <div
   364→                className={`absolute left-6 top-1/2 -translate-y-1/2 pointer-events-none text-lg font-bold transition-opacity duration-500 ${
   365→                  isFading ? "opacity-0" : "opacity-100"
   366→                }`}
   367→                aria-hidden="true"
   368→              >
   369→                <span className="text-muted-foreground/40">
   370→                  {displayedPlaceholder}
   371→                </span>
   372→                <span className="inline-block w-0.5 h-5 bg-mauve/60 ml-0.5 align-middle animate-pulse" />
   373→              </div>
   374→            )}
   375→          </div>
   376→        </div>
   377→
   378→        {/* ─── Source links ─── */}
   379→        <div className="mb-8">
   380→          <label className="block text-sm font-bold mb-3 text-muted-foreground uppercase tracking-wider">
   381→            <LinkIcon className="w-4 h-4 inline mr-1.5 -mt-0.5" />
   382→            {tx.create.links}
   383→          </label>
   384→          <div className="flex gap-2">
   385→            <input
   386→              type="text"
   387→              value={linkInput}
   388→              onChange={(e) => setLinkInput(e.target.value)}
   389→              placeholder={tx.create.linkPlaceholder}
   390→              className="flex-1 px-5 py-3.5 rounded-2xl bg-night border border-border text-foreground font-semibold placeholder:text-muted-foreground/50 focus:outline-none focus:border-mauve focus:ring-2 focus:ring-mauve/20 transition-all duration-300"
   391→              onKeyDown={(e) => {
   392→                if (e.key === "Enter") {
   393→                  e.preventDefault();
   394→                  addLink();
   395→                }
   396→              }}
   397→              disabled={loading}
   398→            />
   399→            <button
   400→              onClick={addLink}
   401→              disabled={!linkInput.trim() || loading}
   402→              className="px-5 py-3.5 rounded-2xl bg-mauve/20 text-mauve-light font-bold hover:bg-mauve/30 transition-all duration-200 disabled:opacity-40 disabled:cursor-not-allowed cursor-pointer"
   403→            >
   404→              <Plus className="w-5 h-5" />
   405→            </button>
   406→          </div>
   407→
   408→          {links.length > 0 && (
   409→            <div className="mt-3 space-y-2">
   410→              {links.map((link, i) => (
   411→                <div
   412→                  key={link}
   413→                  className="flex items-center gap-2 px-4 py-2.5 rounded-xl bg-night/50 border border-border text-sm transition-all duration-200"
   414→                >
   415→                  <LinkIcon className="w-3.5 h-3.5 text-mauve-light flex-shrink-0" />
   416→                  <span className="truncate flex-1 text-muted-foreground">
   417→                    {link}
   418→                  </span>
   419→                  <button
   420→                    onClick={() => removeLink(i)}
   421→                    className="p-1.5 hover:bg-white/10 rounded-lg transition-colors duration-200 cursor-pointer"
   422→                  >
   423→                    <X className="w-3.5 h-3.5 text-muted-foreground" />
   424→                  </button>
   425→                </div>
   426→              ))}
   427→            </div>
   428→          )}
   429→        </div>
   430→
   431→        {/* ─── Course language selector ─── */}
   432→        <div className="mb-8">
   433→          <label className="block text-sm font-bold mb-3 text-muted-foreground uppercase tracking-wider">
   434→            <Globe className="w-4 h-4 inline mr-1.5 -mt-0.5" />
   435→            {tx.create.courseLang}
   436→          </label>
   437→          <div className="grid grid-cols-2 gap-3">
   438→            {(["fr", "en"] as const).map((lng, i) => {
   439→              const isSelected = courseLang === lng;
   440→              return (
   441→                <button
   442→                  key={lng}
   443→                  onClick={() => setCourseLang(lng)}
   444→                  className={`relative px-4 py-4 rounded-2xl font-bold text-center cursor-pointer transition-all duration-300 ${
   445→                    isSelected
   446→                      ? "bg-mauve/20 border-2 border-mauve text-mauve-light shadow-lg shadow-mauve/10"
   447→                      : "glass border-2 border-transparent text-muted-foreground hover:text-foreground hover:bg-white/5"
   448→                  }`}
   449→                >
   450→                  <div className="flex items-center justify-center gap-2">
   451→                    <span className="text-xl">
   452→                      {lng === "fr" ? "🇫🇷" : "🇬🇧"}
   453→                    </span>
   454→                    <span className="text-sm md:text-base">{courseLangLabels[i]}</span>
   455→                  </div>
   456→                  {isSelected && (
   457→                    <div className="absolute -top-1.5 -right-1.5 w-3.5 h-3.5 rounded-full bg-mauve border-2 border-night" />
   458→                  )}
   459→                </button>
   460→              );
   461→            })}
   462→          </div>
   463→        </div>
   464→
   465→        {/* ─── Level selector ─── */}
   466→        {!isRandomTopic && (
   467→          <div className="mb-8">
   468→            <label className="block text-sm font-bold mb-3 text-muted-foreground uppercase tracking-wider">
   469→              {tx.create.level}
   470→            </label>
   471→            <div className="grid grid-cols-3 gap-3">
   472→              {tx.create.levels.map((levelName, i) => {
   473→                const isSelected = selectedLevel === i;
   474→                return (
   475→                  <button
   476→                    key={i}
   477→                    onClick={() => setSelectedLevel(i)}
   478→                    className={`relative px-4 py-4 rounded-2xl font-bold text-center cursor-pointer transition-all duration-300 ${
   479→                      isSelected
   480→                        ? "bg-mauve/20 border-2 border-mauve text-mauve-light shadow-lg shadow-mauve/10"
   481→                        : "glass border-2 border-transparent text-muted-foreground hover:text-foreground hover:bg-white/5"
   482→                    }`}
   483→                  >
   484→                    <span className="text-sm md:text-base">{levelName}</span>
   485→                    {isSelected && (
   486→                      <div className="absolute -top-1.5 -right-1.5 w-3.5 h-3.5 rounded-full bg-mauve border-2 border-night" />
   487→                    )}
   488→                  </button>
   489→                );
   490→              })}
   491→            </div>
   492→          </div>
   493→        )}
   494→        {isRandomTopic && (
   495→          <div className="mb-8 flex items-center gap-2">
   496→            <span className="text-xs font-bold text-muted-foreground uppercase tracking-wider">{tx.create.level}</span>
   497→            <span className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-mauve/15 border border-mauve/25 text-xs font-bold text-mauve-light">
   498→              {tx.create.levels[0]}
   499→            </span>
   500→          </div>
   501→        )}
   502→
   503→        {/* ─── Trial/Grace banner ─── */}
   504→        {!hasSubscription && !inGracePeriod && (
   505→          <div className={`mb-6 p-4 rounded-2xl border text-center transition-all duration-300 ${
   506→            trialExpired || coursesRemaining <= 0
   507→              ? "bg-destructive/10 border-destructive/30"
