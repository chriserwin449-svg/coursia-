     1→"use client";
     2→
     3→import {
     4→  Check,
     5→  Crown,
     6→  Zap,
     7→  AlertTriangle,
     8→  Loader2,
     9→  Clock,
    10→  ShieldAlert,
    11→  Gift,
    12→  Lock,
    13→} from "lucide-react";
    14→import { useAppStore } from "@/lib/store";
    15→import { t } from "@/lib/i18n";
    16→import { trackEvent } from "@/lib/analytics";
    17→import { useState, useMemo, useEffect, useCallback, useRef } from "react";
    18→// PayPal buttons removed — using custom CTA buttons with redirect flow
    19→
    20→export default function OffersPage() {
    21→  const lang = useAppStore((s) => s.lang);
    22→  const userId = useAppStore((s) => s.userId);
    23→  const isAuthenticated = useAppStore((s) => s.isAuthenticated);
    24→  const setView = useAppStore((s) => s.setView);
    25→  const tx = t(lang);
    26→
    27→  const [trialExpired, setTrialExpired] = useState(false);
    28→  const [inTrial, setInTrial] = useState(false);
    29→  const [trialDaysRemaining, setTrialDaysRemaining] = useState(0);
    30→  const [trialCoursesGenerated, setTrialCoursesGenerated] = useState(0);
    31→  const [trialCoursesMax] = useState(3);
    32→  const [isSubscribed, setIsSubscribed] = useState(false);
    33→  const [subscriptionPlan, setSubscriptionPlan] = useState<string>("");
    34→  const [inGracePeriod, setInGracePeriod] = useState(false);
    35→  const [graceDaysRemaining, setGraceDaysRemaining] = useState(0);
    36→  const [graceExpired, setGraceExpired] = useState(false);
    37→  const [showRenewalReminder, setShowRenewalReminder] = useState(false);
    38→  const [renewalDaysRemaining, setRenewalDaysRemaining] = useState(0);
    39→  const [renewalUrgency, setRenewalUrgency] = useState<string>("none");
    40→  const [timeRemainingMs, setTimeRemainingMs] = useState<number | undefined>();
    41→  const [firstName, setFirstName] = useState<string>("");
    42→  const [loadingPlan, setLoadingPlan] = useState<string | null>(null);
    43→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
    44→
    45→  // PayPal payment flow states
    46→  const [paymentRequestId, setPaymentRequestId] = useState<string | null>(null);
    47→  const [paypalOrderId, setPaypalOrderId] = useState<string | null>(null);
    48→  const [paymentSuccess, setPaymentSuccess] = useState(false);
    49→  const [paymentProcessing, setPaymentProcessing] = useState(false);
    50→
    51→  // Countdown timer for last 24 hours
    52→  const [countdown, setCountdown] = useState({ hours: 0, minutes: 0, seconds: 0 });
    53→  const countdownIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
    54→
    55→  // PayPal is always assumed configured — the checkout API will handle errors if not
    56→  const isPaypalConfigured = true;
    57→
    58→  // Handle checkout — create order and redirect to PayPal
    59→  const handleCheckout = useCallback(async (plan: string) => {
    60→    if (!isAuthenticated || !userId) {
    61→      setView("auth");
    62→      return;
    63→    }
    64→    if (loadingPlan) return;
    65→
    66→    setLoadingPlan(plan);
    67→    setCheckoutError(null);
    68→
    69→    try {
    70→      const res = await fetch("/api/subscription/checkout", {
    71→        method: "POST",
    72→        headers: { "Content-Type": "application/json" },
    73→        body: JSON.stringify({ plan, userId }),
    74→      });
    75→
    76→      const data = await res.json();
    77→
    78→      if (!res.ok) {
    79→        setCheckoutError(data.error || (lang === "fr" ? "Erreur lors du paiement" : "Payment failed"));
    80→        setLoadingPlan(null);
    81→        return;
    82→      }
    83→
    84→      if (data.requestId) setPaymentRequestId(data.requestId);
    85→      setPaypalOrderId(data.orderId);
    86→
    87→      // Redirect to PayPal for approval
    88→      if (data.approveUrl) {
    89→        trackEvent({ name: "payment_init", properties: { plan } });
    90→        window.location.href = data.approveUrl;
    91→      } else {
    92→        setCheckoutError(lang === "fr" ? "Lien PayPal indisponible." : "PayPal link unavailable.");
    93→        setLoadingPlan(null);
    94→      }
    95→    } catch {
    96→      setCheckoutError(lang === "fr" ? "Erreur de connexion. Réessaie." : "Connection error. Please try again.");
    97→      setLoadingPlan(null);
    98→    }
    99→  }, [isAuthenticated, userId, lang, setView, loadingPlan]);
   100→
   101→  // Check paywall & subscription status
   102→  useEffect(() => {
   103→    // Track when user views the pricing page
   104→    trackEvent({ name: "pricing_viewed" });
   105→
   106→    const checkStatus = async () => {
   107→      try {
   108→        const headers: Record<string, string> = {};
   109→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   110→
   111→        const res = await fetch("/api/courses/paywall-status", { headers });
   112→        const data = await res.json();
   113→
   114→        if (data.firstName) setFirstName(data.firstName);
   115→
   116→        if (data.inTrial) {
   117→          setInTrial(true);
   118→          setTrialDaysRemaining(data.trialDaysRemaining || 0);
   119→          setTrialCoursesGenerated(data.trialCoursesGenerated || 0);
   120→        } else if (data.showPaywall && data.paywallReason === "trial_expired") {
   121→          setTrialExpired(true);
   122→        }
   123→
   124→        if (data.hasSubscription && data.subscriptionStatus === "active") {
   125→          setIsSubscribed(true);
   126→          setSubscriptionPlan(data.subscriptionPlan || "monthly");
   127→          setTrialExpired(false);
   128→
   129→          if (data.showRenewalReminder) {
   130→            setShowRenewalReminder(true);
   131→            setRenewalDaysRemaining(data.renewalDaysRemaining || 0);
   132→            setRenewalUrgency(data.renewalUrgency || "none");
   133→            setTimeRemainingMs(data.timeRemainingMs);
   134→          }
   135→        }
   136→
   137→        if (data.inGracePeriod) {
   138→          setInGracePeriod(true);
   139→          setGraceDaysRemaining(data.graceDaysRemaining || 0);
   140→        } else if (data.showPaywall && data.paywallReason === "grace_expired") {
   141→          setGraceExpired(true);
   142→        }
   143→      } catch {
   144→        // silently fail
   145→      }
   146→    };
   147→    checkStatus();
   148→  }, [userId]);
   149→
   150→  // Countdown timer for last 24 hours
   151→  useEffect(() => {
   152→    if (renewalUrgency !== "last24hours" || !timeRemainingMs) {
   153→      if (countdownIntervalRef.current) {
   154→        clearInterval(countdownIntervalRef.current);
   155→        countdownIntervalRef.current = null;
   156→      }
   157→      return;
   158→    }
   159→
   160→    const updateCountdown = () => {
   161→      const now = Date.now();
   162→      const remaining = Math.max(0, timeRemainingMs - (now - ((window as unknown as Record<string, number>).__countdownStart || now)));
   163→
   164→      if (remaining <= 0) {
   165→        if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   166→        setCountdown({ hours: 0, minutes: 0, seconds: 0 });
   167→        return;
   168→      }
   169→
   170→      const totalSeconds = Math.floor(remaining / 1000);
   171→      const hours = Math.floor(totalSeconds / 3600);
   172→      const minutes = Math.floor((totalSeconds % 3600) / 60);
   173→      const seconds = totalSeconds % 60;
   174→      setCountdown({ hours, minutes, seconds });
   175→    };
   176→
   177→    (window as unknown as Record<string, number>).__countdownStart = Date.now();
   178→
   179→    const fetchFreshTime = async () => {
   180→      try {
   181→        const headers: Record<string, string> = {};
   182→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   183→        const res = await fetch("/api/courses/paywall-status", { headers });
   184→        const data = await res.json();
   185→        if (data.timeRemainingMs) {
   186→          setTimeRemainingMs(data.timeRemainingMs);
   187→          (window as unknown as Record<string, number>).__countdownStart = Date.now();
   188→        }
   189→      } catch { /* ignore */ }
   190→    };
   191→
   192→    updateCountdown();
   193→    countdownIntervalRef.current = setInterval(updateCountdown, 1000);
   194→    const refreshInterval = setInterval(fetchFreshTime, 30_000);
   195→
   196→    return () => {
   197→      if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   198→      clearInterval(refreshInterval);
   199→    };
   200→  }, [renewalUrgency, timeRemainingMs, userId]);
   201→
   202→  // Get renewal reminder message
   203→  const getRenewalMessage = useMemo(() => {
   204→    const name = firstName || (lang === "fr" ? "Bonjour" : "Hey");
   205→    const isAnnual = subscriptionPlan === "annual";
   206→
   207→    if (renewalUrgency === "last24hours") {
   208→      const key = isAnnual ? "renewalAnnual24hCountdown" : "renewalMonthly24hCountdown";
   209→      return tx.offers[key]
   210→        .replace("{name}", name)
   211→        .replace("{hours}", String(countdown.hours))
   212→        .replace("{minutes}", String(countdown.minutes).padStart(2, "0"))
   213→        .replace("{seconds}", String(countdown.seconds).padStart(2, "0"));
   214→    }
   215→
   216→    const keyMap: Record<string, string> = isAnnual
   217→      ? {
   218→          "1month": "renewalAnnual1Month",
   219→          "2weeks": "renewalAnnual2Weeks",
   220→          "1week": "renewalAnnual1Week",
   221→          "3days": "renewalAnnual3Days",
   222→          "24hours": "renewalAnnual24Hours",
   223→        }
   224→      : {
   225→          "1week": "renewalMonthly1Week",
   226→          "3days": "renewalMonthly3Days",
   227→          "24hours": "renewalMonthly24Hours",
   228→        };
   229→
   230→    const key = keyMap[renewalUrgency];
   231→    if (key && (tx.offers as Record<string, string>)[key]) {
   232→      return (tx.offers as Record<string, string>)[key].replace("{name}", name);
   233→    }
   234→
   235→    return null;
   236→  }, [renewalUrgency, firstName, subscriptionPlan, lang, countdown, tx.offers]);
   237→
   238→  const getUrgencyColor = () => {
   239→    if (renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days") {
   240→      return "bg-red-500/10 border-red-500/30 text-red-200";
   241→    }
   242→    return "bg-amber-500/10 border-amber-500/30 text-amber-200";
   243→  };
   244→
   245→  // Check URL params for payment success
   246→  useEffect(() => {
   247→    const params = new URLSearchParams(window.location.search);
   248→    const paymentStatus = params.get("payment");
   249→    const planParam = params.get("plan");
   250→
   251→    if (paymentStatus === "success" && planParam) {
   252→      window.history.replaceState({}, "", "/");
   253→      // Refresh status to show subscription is active
   254→      const checkAfterCheckout = async () => {
   255→        try {
   256→          const headers: Record<string, string> = {};
   257→          if (userId) headers["Authorization"] = `Bearer ${userId}`;
   258→          const res = await fetch("/api/courses/paywall-status", { headers });
   259→          const data = await res.json();
   260→          if (data.hasSubscription && data.subscriptionStatus === "active") {
   261→            setIsSubscribed(true);
   262→            setSubscriptionPlan(planParam);
   263→            setTrialExpired(false);
   264→            setInGracePeriod(false);
   265→            setGraceExpired(false);
   266→            setShowRenewalReminder(false);
   267→          }
   268→        } catch { /* silent */ }
   269→      };
   270→      setTimeout(checkAfterCheckout, 2000);
   271→    } else if (paymentStatus === "cancelled") {
   272→      window.history.replaceState({}, "", "/");
   273→      // Use setTimeout to avoid calling setState directly in effect
   274→      setTimeout(() => {
   275→        setCheckoutError(lang === "fr" ? "Paiement annulé. Tu peux réessayer." : "Payment cancelled. You can try again.");
   276→      }, 0);
   277→    }
   278→  }, [userId, lang]);
   279→
   280→  const suffix = (n: number) => (n > 1 ? (lang === "fr" ? "s" : "s") : "");
   281→
   282→  const cannotRenewMessage = useMemo(() => {
   283→    if (isSubscribed && !showRenewalReminder) {
   284→      const name = firstName || "";
   285→      return tx.offers.cannotRenewEarly.replace("{name}", name || (lang === "fr" ? "Bonjour" : "Hey"));
   286→    }
   287→    return null;
   288→  }, [isSubscribed, showRenewalReminder, firstName, lang, tx.offers]);
   289→
   290→  // Button disabled logic
   291→  const isButtonDisabled = (plan: string) =>
   292→    loadingPlan === plan ||
   293→    paymentProcessing ||
   294→    (isSubscribed && !showRenewalReminder && !inGracePeriod && !graceExpired);
   295→
   296→  return (
   297→    <>
   298→    <div className="min-h-screen bg-night px-4 sm:px-6 md:px-10 pt-14 sm:pt-20 pb-4 sm:pb-6 md:pb-10 lg:pb-14 md:pt-24">
   299→      {/* Background glow */}
   300→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   301→        <div className="absolute top-1/3 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[800px] h-[800px] bg-mauve/5 rounded-full blur-[150px]" />
   302→      </div>
   303→
   304→      <div className="relative z-10 max-w-6xl mx-auto">
   305→        {/* ===== HEADER ===== */}
   306→        <div className="text-center mb-12">
   307→          <h1 className="text-4xl sm:text-5xl md:text-6xl font-extrabold mb-4">
   308→            <span className="gradient-text">{tx.offers.title}</span>
   309→          </h1>
   310→          <p className="text-lg sm:text-xl text-muted-foreground max-w-2xl mx-auto">
   311→            {tx.offers.subtitle}
   312→          </p>
   313→        </div>
   314→
   315→        {/* ===== PAYMENT SUCCESS BANNER ===== */}
   316→        {paymentSuccess && (
   317→          <div className="max-w-2xl mx-auto mb-8 space-y-3">
   318→            <div className="flex items-start gap-3 p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   319→              <Check className="w-6 h-6 text-emerald-400 flex-shrink-0 mt-0.5" />
   320→              <div>
   321→                <p className="text-base font-bold text-emerald-300">
   322→                  {lang === "fr" ? "Paiement réussi ! 🎉" : "Payment successful! 🎉"}
   323→                </p>
   324→                <p className="text-sm text-emerald-400/70 mt-1">
   325→                  {lang === "fr"
   326→                    ? "Ton abonnement est maintenant actif. Tu as accès à tous les cours."
   327→                    : "Your subscription is now active. You have access to all courses."}
   328→                </p>
   329→              </div>
   330→            </div>
   331→          </div>
   332→        )}
   333→
   334→        {/* ===== PAYMENT PROCESSING BANNER ===== */}
   335→        {paymentProcessing && !paymentSuccess && (
   336→          <div className="max-w-2xl mx-auto mb-8">
   337→            <div className="flex items-center gap-3 p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   338→              <Loader2 className="w-5 h-5 text-mauve-light animate-spin" />
   339→              <p className="text-sm text-mauve-light font-medium">
   340→                {lang === "fr" ? "Confirmation du paiement en cours..." : "Confirming payment..."}
   341→              </p>
   342→            </div>
   343→          </div>
   344→        )}
   345→
   346→        {/* ===== STATUS BANNERS ===== */}
   347→        <div className="max-w-2xl mx-auto mb-8 space-y-3">
   348→          {/* Personalized renewal reminder */}
   349→          {showRenewalReminder && getRenewalMessage && (
   350→            <div className={`flex items-start gap-3 p-4 sm:p-5 rounded-2xl border animate-fade-in ${getUrgencyColor()}`}>
   351→              <Clock className={`w-5 h-5 flex-shrink-0 mt-0.5 ${renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days" ? "text-red-400" : "text-amber-400"}`} />
   352→              <div>
   353→                <p className="text-xs font-bold uppercase tracking-wider mb-1 opacity-70">
   354→                  {tx.offers.subscriptionEndingSoon}
   355→                </p>
   356→                <p className="text-sm sm:text-base font-medium">{getRenewalMessage}</p>
   357→                {renewalUrgency === "last24hours" && (
   358→                  <p className="text-2xl sm:text-3xl font-extrabold mt-2 tabular-nums tracking-wide">
   359→                    {String(countdown.hours).padStart(2, "0")}:{String(countdown.minutes).padStart(2, "0")}:{String(countdown.seconds).padStart(2, "0")}
   360→                  </p>
   361→                )}
   362→              </div>
   363→            </div>
   364→          )}
   365→
   366→          {/* Grace period */}
   367→          {inGracePeriod && (
   368→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   369→              <Clock className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   370→              <div>
   371→                <p className="text-sm sm:text-base text-amber-200 font-medium">
   372→                  {tx.offers.gracePeriod.replace("{days}", String(graceDaysRemaining)).replace("{suffix}", suffix(graceDaysRemaining))}
   373→                </p>
   374→                <p className="text-xs text-amber-300/70 mt-1">{tx.offers.graceReadonly}</p>
   375→              </div>
   376→            </div>
   377→          )}
   378→
   379→          {/* Grace expired */}
   380→          {graceExpired && (
   381→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   382→              <ShieldAlert className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   383→              <p className="text-sm sm:text-base text-red-200 font-medium">{tx.offers.gracePeriodExpired}</p>
   384→            </div>
   385→          )}
   386→
   387→          {/* Trial active */}
   388→          {inTrial && (
   389→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   390→              <Gift className="w-5 h-5 text-mauve-light flex-shrink-0 mt-0.5" />
   391→              <div>
   392→                <p className="text-sm sm:text-base text-mauve-light font-medium">
   393→                  {tx.offers.daysRemaining.replace("{days}", String(trialDaysRemaining)).replace("{suffix}", suffix(trialDaysRemaining))}
   394→                </p>
   395→                <p className="text-xs text-mauve-light/70 mt-1">
   396→                  {tx.offers.coursesRemaining.replace("{count}", String(trialCoursesMax - trialCoursesGenerated)).replace("{suffix}", suffix(trialCoursesMax - trialCoursesGenerated))}
   397→                </p>
   398→              </div>
   399→            </div>
   400→          )}
   401→
   402→          {/* Trial expired */}
   403→          {trialExpired && (
   404→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   405→              <AlertTriangle className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   406→              <p className="text-sm sm:text-base text-amber-200 font-medium">{tx.offers.trialExpired}</p>
   407→            </div>
   408→          )}
   409→
   410→          {/* Already subscribed (not ending) */}
   411→          {isSubscribed && !showRenewalReminder && (
   412→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   413→              <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   414→              <div>
   415→                <p className="text-sm sm:text-base text-emerald-200 font-medium">{tx.offers.subscribed}</p>
   416→                {cannotRenewMessage && (
   417→                  <p className="text-xs text-emerald-300/70 mt-1">{cannotRenewMessage}</p>
   418→                )}
   419→              </div>
   420→            </div>
   421→          )}
   422→
   423→          {/* Checkout error */}
   424→          {checkoutError && (
   425→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   426→              <AlertTriangle className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   427→              <p className="text-sm sm:text-base text-red-200 font-medium">{checkoutError}</p>
   428→            </div>
   429→          )}
   430→        </div>
   431→
   432→        {/* ===== PRICING CARDS ===== */}
   433→        {!paymentSuccess && (
   434→          <div
   435→            className={`grid gap-6 lg:gap-8 items-start mb-20 grid-cols-1 md:grid-cols-2 max-w-4xl mx-auto`}
   436→          >
   437→            {/* MONTHLY PLAN */}
   438→            <div className="pricing-card-float monthly-card-glow glass rounded-3xl p-5 sm:p-8 flex flex-col hover:border-mauve/30 transition-all duration-300">
   439→              <div className="mb-6">
   440→                <div className="flex items-center gap-3 mb-2">
   441→                  <div className="w-11 h-11 sm:w-12 sm:h-12 rounded-2xl bg-mauve/10 flex items-center justify-center">
   442→                    <Zap className="w-6 h-6 text-mauve-light" />
   443→                  </div>
   444→                  <h3 className="text-lg sm:text-xl font-bold">{tx.landing.pricing.monthly.name}</h3>
   445→                </div>
   446→                <p className="text-muted-foreground text-sm">{tx.landing.pricing.monthly.desc}</p>
   447→              </div>
   448→              <div className="mb-6">
   449→                <span className="text-3xl sm:text-4xl font-extrabold">{tx.landing.pricing.monthly.price}</span>
   450→                <span className="text-lg text-muted-foreground">{tx.landing.pricing.monthly.period}</span>
   451→              </div>
   452→              <ul className="flex-1 space-y-2 sm:space-y-3 mb-8">
   453→                {tx.landing.pricing.monthly.features.map((f) => (
   454→                  <li key={f} className="flex items-start gap-3">
   455→                    <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   456→                    <span className="text-sm text-muted-foreground">{f}</span>
   457→                  </li>
   458→                ))}
   459→              </ul>
   460→
   461→              {/* CTA Button */}
   462→              {isButtonDisabled("monthly") ? (
   463→                <button
   464→                  disabled
   465→                  className="w-full py-3.5 sm:py-4 rounded-full bg-gradient-to-r from-mauve to-mauve-dark text-white font-bold opacity-50 cursor-not-allowed flex items-center justify-center gap-2"
   466→                >
   467→                  {isSubscribed && !showRenewalReminder && !inGracePeriod && !graceExpired
   468→                    ? lang === "fr" ? "Plan Actuel" : "Current Plan"
   469→                    : <>
