     1→"use client";
     2→
     3→import { Check, Crown, Zap, AlertTriangle, Loader2, Clock, ShieldAlert, Gift, CreditCard } from "lucide-react";
     4→import { useAppStore } from "@/lib/store";
     5→import { t } from "@/lib/i18n";
     6→import { useState, useMemo, useEffect, useCallback, useRef } from "react";
     7→
     8→export default function OffersPage() {
     9→  const lang = useAppStore((s) => s.lang);
    10→  const userId = useAppStore((s) => s.userId);
    11→  const isAuthenticated = useAppStore((s) => s.isAuthenticated);
    12→  const setView = useAppStore((s) => s.setView);
    13→  const tx = t(lang);
    14→
    15→  const [trialExpired, setTrialExpired] = useState(false);
    16→  const [inTrial, setInTrial] = useState(false);
    17→  const [trialDaysRemaining, setTrialDaysRemaining] = useState(0);
    18→  const [trialCoursesGenerated, setTrialCoursesGenerated] = useState(0);
    19→  const [trialCoursesMax] = useState(3);
    20→  const [isSubscribed, setIsSubscribed] = useState(false);
    21→  const [subscriptionPlan, setSubscriptionPlan] = useState<string>("");
    22→  const [inGracePeriod, setInGracePeriod] = useState(false);
    23→  const [graceDaysRemaining, setGraceDaysRemaining] = useState(0);
    24→  const [graceExpired, setGraceExpired] = useState(false);
    25→  const [showRenewalReminder, setShowRenewalReminder] = useState(false);
    26→  const [renewalDaysRemaining, setRenewalDaysRemaining] = useState(0);
    27→  const [renewalUrgency, setRenewalUrgency] = useState<string>("none");
    28→  const [timeRemainingMs, setTimeRemainingMs] = useState<number | undefined>();
    29→  const [firstName, setFirstName] = useState<string>("");
    30→  const [loadingPlan, setLoadingPlan] = useState<string | null>(null);
    31→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
    32→
    33→  // Fondeka/Chariow payment flow states
    34→  const [paymentRequestId, setPaymentRequestId] = useState<string | null>(null);
    35→  const [showPaymentSteps, setShowPaymentSteps] = useState(false);
    36→  const [confirmingPayment, setConfirmingPayment] = useState(false);
    37→  const [paymentConfirmed, setPaymentConfirmed] = useState(false);
    38→
    39→  // Countdown timer for last 24 hours
    40→  const [countdown, setCountdown] = useState({ hours: 0, minutes: 0, seconds: 0 });
    41→  const countdownIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
    42→
    43→  // Check paywall & subscription status
    44→  useEffect(() => {
    45→    const checkStatus = async () => {
    46→      try {
    47→        const headers: Record<string, string> = {};
    48→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
    49→
    50→        const res = await fetch("/api/courses/paywall-status", { headers });
    51→        const data = await res.json();
    52→
    53→        // First name
    54→        if (data.firstName) setFirstName(data.firstName);
    55→
    56→        // Trial
    57→        if (data.inTrial) {
    58→          setInTrial(true);
    59→          setTrialDaysRemaining(data.trialDaysRemaining || 0);
    60→          setTrialCoursesGenerated(data.trialCoursesGenerated || 0);
    61→        } else if (data.showPaywall && data.paywallReason === "trial_expired") {
    62→          setTrialExpired(true);
    63→        }
    64→
    65→        // Subscription
    66→        if (data.hasSubscription && data.subscriptionStatus === "active") {
    67→          setIsSubscribed(true);
    68→          setSubscriptionPlan(data.subscriptionPlan || "monthly");
    69→          setTrialExpired(false);
    70→
    71→          if (data.showRenewalReminder) {
    72→            setShowRenewalReminder(true);
    73→            setRenewalDaysRemaining(data.renewalDaysRemaining || 0);
    74→            setRenewalUrgency(data.renewalUrgency || "none");
    75→            setTimeRemainingMs(data.timeRemainingMs);
    76→          }
    77→        }
    78→
    79→        // Grace period
    80→        if (data.inGracePeriod) {
    81→          setInGracePeriod(true);
    82→          setGraceDaysRemaining(data.graceDaysRemaining || 0);
    83→        } else if (data.showPaywall && data.paywallReason === "grace_expired") {
    84→          setGraceExpired(true);
    85→        }
    86→      } catch {
    87→        // silently fail
    88→      }
    89→    };
    90→    checkStatus();
    91→  }, [userId]);
    92→
    93→  // Countdown timer for last 24 hours
    94→  useEffect(() => {
    95→    if (renewalUrgency !== "last24hours" || !timeRemainingMs) {
    96→      if (countdownIntervalRef.current) {
    97→        clearInterval(countdownIntervalRef.current);
    98→        countdownIntervalRef.current = null;
    99→      }
   100→      return;
   101→    }
   102→
   103→    const updateCountdown = () => {
   104→      const now = Date.now();
   105→      // Recalculate from subscription end date (stored initially)
   106→      const remaining = Math.max(0, timeRemainingMs - (now - ((window as unknown as Record<string, number>).__countdownStart || now)));
   107→
   108→      if (remaining <= 0) {
   109→        if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   110→        setCountdown({ hours: 0, minutes: 0, seconds: 0 });
   111→        return;
   112→      }
   113→
   114→      const totalSeconds = Math.floor(remaining / 1000);
   115→      const hours = Math.floor(totalSeconds / 3600);
   116→      const minutes = Math.floor((totalSeconds % 3600) / 60);
   117→      const seconds = totalSeconds % 60;
   118→      setCountdown({ hours, minutes, seconds });
   119→    };
   120→
   121→    // Store the start time for calculations
   122→    (window as unknown as Record<string, number>).__countdownStart = Date.now();
   123→
   124→    // Fetch fresh timeRemainingMs periodically to stay in sync
   125→    const fetchFreshTime = async () => {
   126→      try {
   127→        const headers: Record<string, string> = {};
   128→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   129→        const res = await fetch("/api/courses/paywall-status", { headers });
   130→        const data = await res.json();
   131→        if (data.timeRemainingMs) {
   132→          setTimeRemainingMs(data.timeRemainingMs);
   133→          (window as unknown as Record<string, number>).__countdownStart = Date.now();
   134→        }
   135→      } catch { /* ignore */ }
   136→    };
   137→
   138→    updateCountdown();
   139→    countdownIntervalRef.current = setInterval(() => {
   140→      updateCountdown();
   141→    }, 1000);
   142→
   143→    // Refresh from server every 30 seconds to prevent drift
   144→    const refreshInterval = setInterval(fetchFreshTime, 30_000);
   145→
   146→    return () => {
   147→      if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   148→      clearInterval(refreshInterval);
   149→    };
   150→  }, [renewalUrgency, timeRemainingMs, userId]);
   151→
   152→  // Get the renewal reminder message based on urgency
   153→  const getRenewalMessage = useMemo(() => {
   154→    const name = firstName || (lang === "fr" ? "Bonjour" : "Hey");
   155→    const isAnnual = subscriptionPlan === "annual";
   156→
   157→    if (renewalUrgency === "last24hours") {
   158→      const key = isAnnual ? "renewalAnnual24hCountdown" : "renewalMonthly24hCountdown";
   159→      return tx.offers[key]
   160→        .replace("{name}", name)
   161→        .replace("{hours}", String(countdown.hours))
   162→        .replace("{minutes}", String(countdown.minutes).padStart(2, "0"))
   163→        .replace("{seconds}", String(countdown.seconds).padStart(2, "0"));
   164→    }
   165→
   166→    const keyMap: Record<string, string> = isAnnual
   167→      ? {
   168→          "1month": "renewalAnnual1Month",
   169→          "2weeks": "renewalAnnual2Weeks",
   170→          "1week": "renewalAnnual1Week",
   171→          "3days": "renewalAnnual3Days",
   172→          "24hours": "renewalAnnual24Hours",
   173→        }
   174→      : {
   175→          "1week": "renewalMonthly1Week",
   176→          "3days": "renewalMonthly3Days",
   177→          "24hours": "renewalMonthly24Hours",
   178→        };
   179→
   180→    const key = keyMap[renewalUrgency];
   181→    if (key && (tx.offers as Record<string, string>)[key]) {
   182→      return (tx.offers as Record<string, string>)[key].replace("{name}", name);
   183→    }
   184→
   185→    return null;
   186→  }, [renewalUrgency, firstName, subscriptionPlan, lang, countdown, tx.offers]);
   187→
   188→  // Determine the urgency color
   189→  const getUrgencyColor = () => {
   190→    if (renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days") {
   191→      return "bg-red-500/10 border-red-500/30 text-red-200";
   192→    }
   193→    if (renewalUrgency === "1week") {
   194→      return "bg-amber-500/10 border-amber-500/30 text-amber-200";
   195→    }
   196→    return "bg-amber-500/10 border-amber-500/30 text-amber-200";
   197→  };
   198→
   199→  // Check URL params for payment=pending (user returned from Chariow)
   200→  useEffect(() => {
   201→    const params = new URLSearchParams(window.location.search);
   202→    const paymentStatus = params.get("payment");
   203→    const requestId = params.get("request_id");
   204→
   205→    if (paymentStatus === "pending" && requestId) {
   206→      setPaymentRequestId(requestId);
   207→      setShowPaymentSteps(true);
   208→      // Clean URL
   209→      window.history.replaceState({}, "", "/");
   210→    } else if (paymentStatus === "success") {
   211→      // Also handle the old success flow
   212→      const checkAfterCheckout = async () => {
   213→        try {
   214→          const headers: Record<string, string> = {};
   215→          if (userId) headers["Authorization"] = `Bearer ${userId}`;
   216→          const res = await fetch("/api/courses/paywall-status", { headers });
   217→          const data = await res.json();
   218→          if (data.hasSubscription && data.subscriptionStatus === "active") {
   219→            setIsSubscribed(true);
   220→            setTrialExpired(false);
   221→            setInGracePeriod(false);
   222→            setGraceExpired(false);
   223→            setShowRenewalReminder(false);
   224→          }
   225→        } catch { /* silent */ }
   226→      };
   227→      setTimeout(checkAfterCheckout, 2000);
   228→      window.history.replaceState({}, "", "/");
   229→    }
   230→  }, [userId]);
   231→
   232→  // Handle checkout — calls API then opens Chariow link in new tab
   233→  const handleCheckout = useCallback(
   234→    async (plan: "monthly" | "annual") => {
   235→      if (!isAuthenticated || !userId) {
   236→        setView("auth");
   237→        return;
   238→      }
   239→
   240→      setLoadingPlan(plan);
   241→      setCheckoutError(null);
   242→
   243→      try {
   244→        const res = await fetch("/api/subscription/checkout", {
   245→          method: "POST",
   246→          headers: { "Content-Type": "application/json" },
   247→          body: JSON.stringify({ plan, userId }),
   248→        });
   249→
   250→        const data = await res.json();
   251→
   252→        if (!res.ok) {
   253→          setCheckoutError(
   254→            data.error || (lang === "fr" ? "Erreur lors de la création du paiement" : "Payment creation failed")
   255→          );
   256→          return;
   257→        }
   258→
   259→        // Store request ID for "I paid" flow
   260→        if (data.requestId) {
   261→          setPaymentRequestId(data.requestId);
   262→        }
   263→
   264→        // Open Chariow payment link in new tab
   265→        if (data.checkoutUrl) {
   266→          window.open(data.checkoutUrl, "_blank");
   267→          // Show payment steps after redirect
   268→          setShowPaymentSteps(true);
   269→        }
   270→      } catch {
   271→        setCheckoutError(
   272→          lang === "fr" ? "Erreur de connexion. Réessaie." : "Connection error. Please try again."
   273→        );
   274→      } finally {
   275→        setLoadingPlan(null);
   276→      }
   277→    },
   278→    [isAuthenticated, userId, lang, setView]
   279→  );
   280→
   281→  // Handle "I paid" confirmation
   282→  const handleConfirmPayment = useCallback(async () => {
   283→    if (!userId || !paymentRequestId) return;
   284→
   285→    setConfirmingPayment(true);
   286→    try {
   287→      const res = await fetch("/api/subscription/confirm", {
   288→        method: "POST",
   289→        headers: { "Content-Type": "application/json" },
   290→        body: JSON.stringify({
   291→          userId,
   292→          requestId: paymentRequestId,
   293→        }),
   294→      });
   295→
   296→      const data = await res.json();
   297→
   298→      if (data.success) {
   299→        setPaymentConfirmed(true);
   300→        // Poll subscription status every 10 seconds
   301→        const pollInterval = setInterval(async () => {
   302→          try {
   303→            const headers: Record<string, string> = {};
   304→            if (userId) headers["Authorization"] = `Bearer ${userId}`;
   305→            const statusRes = await fetch("/api/courses/paywall-status", { headers });
   306→            const statusData = await statusRes.json();
   307→            if (statusData.hasSubscription && statusData.subscriptionStatus === "active") {
   308→              clearInterval(pollInterval);
   309→              setIsSubscribed(true);
   310→              setPaymentConfirmed(false);
   311→              setShowPaymentSteps(false);
   312→              setShowRenewalReminder(false);
   313→            }
   314→          } catch { /* keep polling */ }
   315→        }, 10_000);
   316→
   317→        // Auto-stop polling after 5 minutes
   318→        setTimeout(() => clearInterval(pollInterval), 300_000);
   319→      } else {
   320→        setCheckoutError(data.error || "Confirmation failed");
   321→      }
   322→    } catch {
   323→      setCheckoutError(lang === "fr" ? "Erreur de connexion" : "Connection error");
   324→    } finally {
   325→      setConfirmingPayment(false);
   326→    }
   327→  }, [userId, paymentRequestId, lang]);
   328→
   329→  const suffix = (n: number) => (n > 1 ? (lang === "fr" ? "s" : "s") : "");
   330→
   331→  // Personalized message when subscribed but not ending
   332→  const cannotRenewMessage = useMemo(() => {
   333→    if (isSubscribed && !showRenewalReminder) {
   334→      const name = firstName || "";
   335→      return tx.offers.cannotRenewEarly.replace("{name}", name || (lang === "fr" ? "Bonjour" : "Hey"));
   336→    }
   337→    return null;
   338→  }, [isSubscribed, showRenewalReminder, firstName, lang, tx.offers]);
   339→
   340→  return (
   341→    <>
   342→    <div className="min-h-screen bg-night px-4 sm:px-6 md:px-10 pt-14 sm:pt-20 pb-4 sm:pb-6 md:pb-10 lg:pb-14 md:pt-24">
   343→      {/* Background glow */}
   344→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   345→        <div className="absolute top-1/3 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[800px] h-[800px] bg-mauve/5 rounded-full blur-[150px]" />
   346→      </div>
   347→
   348→      <div className="relative z-10 max-w-6xl mx-auto">
   349→        {/* ===== HEADER ===== */}
   350→        <div className="text-center mb-12">
   351→          <h1 className="text-4xl sm:text-5xl md:text-6xl font-extrabold mb-4">
   352→            <span className="gradient-text">{tx.offers.title}</span>
   353→          </h1>
   354→          <p className="text-lg sm:text-xl text-muted-foreground max-w-2xl mx-auto">
   355→            {tx.offers.subtitle}
   356→          </p>
   357→        </div>
   358→
   359→        {/* ===== PAYMENT STEPS (shown after redirect to Chariow) ===== */}
   360→        {showPaymentSteps && !isSubscribed && (
   361→          <div className="max-w-2xl mx-auto mb-8 space-y-3">
   362→            <div className="glass rounded-3xl p-5 sm:p-6 border border-mauve/30 animate-fade-in">
   363→              <div className="flex items-center gap-3 mb-4">
   364→                <div className="w-10 h-10 rounded-xl bg-mauve/20 flex items-center justify-center">
   365→                  <CreditCard className="w-5 h-5 text-mauve-light" />
   366→                </div>
   367→                <h3 className="text-lg font-bold">{tx.offers.paymentPendingTitle}</h3>
   368→              </div>
   369→
   370→              {paymentConfirmed ? (
   371→                <div className="space-y-3">
   372→                  <div className="flex items-start gap-3 p-3 rounded-xl bg-emerald-500/10 border border-emerald-500/30">
   373→                    <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   374→                    <div>
   375→                      <p className="text-sm font-semibold text-emerald-300">{tx.offers.paymentConfirmed}</p>
   376→                      <p className="text-xs text-emerald-400/70 mt-1">{tx.offers.paymentConfirmedDesc}</p>
   377→                    </div>
   378→                  </div>
   379→                  <div className="flex items-center justify-center gap-2 py-2">
   380→                    <Loader2 className="w-4 h-4 text-muted-foreground animate-spin" />
   381→                    <span className="text-xs text-muted-foreground">
   382→                      {lang === "fr" ? "Vérification en cours..." : "Verification in progress..."}
   383→                    </span>
   384→                  </div>
   385→                </div>
   386→              ) : (
   387→                <>
   388→                  <p className="text-sm text-muted-foreground mb-4">{tx.offers.paymentPendingDesc}</p>
   389→                  <div className="space-y-2 mb-5">
   390→                    <p className="text-sm text-foreground/80">{tx.offers.paymentPendingStep1}</p>
   391→                    <p className="text-sm text-foreground/80">{tx.offers.paymentPendingStep2}</p>
   392→                    <p className="text-sm text-foreground/80">{tx.offers.paymentPendingStep3}</p>
   393→                    <p className="text-sm text-muted-foreground">{tx.offers.paymentPendingStep4}</p>
   394→                  </div>
   395→                  <button
   396→                    onClick={handleConfirmPayment}
   397→                    disabled={confirmingPayment}
   398→                    className="w-full py-3.5 rounded-full bg-gradient-to-r from-emerald-500 to-emerald-600 text-white font-bold hover:from-emerald-400 hover:to-emerald-500 transition-all duration-300 flex items-center justify-center gap-2 cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed"
   399→                  >
   400→                    {confirmingPayment ? (
   401→                      <>
   402→                        <Loader2 className="w-5 h-5 animate-spin" />
   403→                        {lang === "fr" ? "Envoi en cours..." : "Sending..."}
   404→                      </>
   405→                    ) : (
   406→                      <>
   407→                        <Check className="w-5 h-5" />
   408→                        {tx.offers.iPaid}
   409→                      </>
   410→                    )}
   411→                  </button>
   412→                </>
   413→              )}
   414→            </div>
   415→          </div>
   416→        )}
   417→
   418→        {/* ===== STATUS BANNERS ===== */}
   419→        <div className="max-w-2xl mx-auto mb-8 space-y-3">
   420→          {/* Personalized renewal reminder (subscribed, ending soon) */}
   421→          {showRenewalReminder && getRenewalMessage && (
   422→            <div className={`flex items-start gap-3 p-4 sm:p-5 rounded-2xl border animate-fade-in ${getUrgencyColor()}`}>
   423→              <Clock className={`w-5 h-5 flex-shrink-0 mt-0.5 ${renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days" ? "text-red-400" : "text-amber-400"}`} />
   424→              <div>
   425→                <p className="text-xs font-bold uppercase tracking-wider mb-1 opacity-70">
   426→                  {tx.offers.subscriptionEndingSoon}
   427→                </p>
   428→                <p className="text-sm sm:text-base font-medium">
   429→                  {getRenewalMessage}
   430→                </p>
   431→                {renewalUrgency === "last24hours" && (
   432→                  <p className="text-2xl sm:text-3xl font-extrabold mt-2 tabular-nums tracking-wide">
   433→                    {String(countdown.hours).padStart(2, "0")}:{String(countdown.minutes).padStart(2, "0")}:{String(countdown.seconds).padStart(2, "0")}
   434→                  </p>
   435→                )}
   436→              </div>
   437→            </div>
   438→          )}
   439→
   440→          {/* Subscription ended, in grace period */}
   441→          {inGracePeriod && (
   442→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   443→              <Clock className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   444→              <div>
   445→                <p className="text-sm sm:text-base text-amber-200 font-medium">
   446→                  {tx.offers.gracePeriod.replace("{days}", String(graceDaysRemaining)).replace("{suffix}", suffix(graceDaysRemaining))}
   447→                </p>
   448→                <p className="text-xs text-amber-300/70 mt-1">{tx.offers.graceReadonly}</p>
   449→              </div>
   450→            </div>
   451→          )}
   452→
   453→          {/* Grace period expired */}
   454→          {graceExpired && (
   455→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   456→              <ShieldAlert className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   457→              <p className="text-sm sm:text-base text-red-200 font-medium">
   458→                {tx.offers.gracePeriodExpired}
   459→              </p>
   460→            </div>
   461→          )}
   462→
   463→          {/* Trial active banner */}
   464→          {inTrial && (
   465→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   466→              <Gift className="w-5 h-5 text-mauve-light flex-shrink-0 mt-0.5" />
   467→              <div>
   468→                <p className="text-sm sm:text-base text-mauve-light font-medium">
   469→                  {tx.offers.daysRemaining.replace("{days}", String(trialDaysRemaining)).replace("{suffix}", suffix(trialDaysRemaining))}
   470→                </p>
   471→                <p className="text-xs text-mauve-light/70 mt-1">
   472→                  {tx.offers.coursesRemaining.replace("{count}", String(trialCoursesMax - trialCoursesGenerated)).replace("{suffix}", suffix(trialCoursesMax - trialCoursesGenerated))}
   473→                </p>
   474→              </div>
   475→            </div>
   476→          )}
   477→
   478→          {/* Trial expired banner */}
   479→          {trialExpired && (
   480→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   481→              <AlertTriangle className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   482→              <p className="text-sm sm:text-base text-amber-200 font-medium">
   483→                {tx.offers.trialExpired}
   484→              </p>
   485→            </div>
   486→          )}
   487→
   488→          {/* Already subscribed banner (not ending) — personalized */}
   489→          {isSubscribed && !showRenewalReminder && (
   490→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   491→              <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   492→              <div>
