   258→   258→      }

   260→   260→      setFetchError(true);

   269→   269→    const chLevel = ch.level ?? 0;
   270→   270→    if (chLevel > (data.maxUnlockedLevel ?? 0)) return true;

   308→   308→      const res = await fetch(`/api/courses/${selectedCourseId}/chapters/${currentChapter.id}/complete`, {

   310→   310→        headers: { "Content-Type": "application/json" },
   311→   311→      });

   314→   314→        if (courseRes.ok) setCourse(await courseRes.json());

   316→   316→      }
   317→   317→      return false;

   350→   350→
   351→   351→  const goToPrev = useCallback(() => {

   364→   364→    setShowCelebration(true);
   365→   365→    setCelebrationMessage(lang === "fr" ? `Félicitations ${userName} ! 🏆` : `Congratulations ${userName}! 🏆`);

   380→   380→        setShowAllMastered(true);

   387→   387→  }, [course, user?.firstName, lang, endStudySession, fetchCourse]);
   388→   388→
   389→   389→  // ── Continue to next level ──
   390→   390→  const handleContinueToNextLevel = useCallback(async () => {
   391→   391→    if (!course || !selectedCourseId) return;
   392→   392→    const nextLevel = maxUnlockedLevel + 1;
   393→   393→    if (nextLevel > 2) return;

   396→   396→    setShowReviewScreen(false);

   402→   402→        headers: { "Content-Type": "application/json" },
   403→   403→        body: JSON.stringify({ level: nextLevel }),
   404→   404→      });

   406→   406→      if (res.ok && data.chapters?.length > 0) {

   408→   408→        await fetchCourse();

   410→   410→        const firstNewChapterIdx = course.chapters.findIndex((ch) => (ch.level ?? 0) === nextLevel);
   411→   411→        if (firstNewChapterIdx >= 0) setCurrentChapterIndex(firstNewChapterIdx);
   412→   412→      }
   413→   413→    } catch {
   414→   414→      // show error
   415→   415→    } finally {
   416→   416→      setIsGeneratingLevel(false);
   417→   417→    }
   418→   418→  }, [course, selectedCourseId, maxUnlockedLevel, fetchCourse, setCurrentChapterIndex]);
   419→   419→
   420→   420→  // ── Stop at current level ──
   421→   421→  const handleStopHere = useCallback(async () => {

   423→   423→    try {
   424→   424→      await fetch(`/api/courses/${selectedCourseId}/stop-level`, {
   425→   425→        method: "POST",
   426→   426→        headers: { "Content-Type": "application/json" },
   427→   427→      });
