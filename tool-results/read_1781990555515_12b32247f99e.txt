    18→    18→// PayPal buttons removed — using custom CTA buttons with redirect flow
    19→    19→

    26→    26→
    27→    27→  const [trialExpired, setTrialExpired] = useState(false);

    35→    35→  const [graceDaysRemaining, setGraceDaysRemaining] = useState(0);
    36→    36→  const [graceExpired, setGraceExpired] = useState(false);
    37→    37→  const [showRenewalReminder, setShowRenewalReminder] = useState(false);
    38→    38→  const [renewalDaysRemaining, setRenewalDaysRemaining] = useState(0);
    39→    39→  const [renewalUrgency, setRenewalUrgency] = useState<string>("none");
    40→    40→  const [timeRemainingMs, setTimeRemainingMs] = useState<number | undefined>();
    41→    41→  const [firstName, setFirstName] = useState<string>("");
    42→    42→  const [loadingPlan, setLoadingPlan] = useState<string | null>(null);

    50→    50→
    51→    51→  // Countdown timer for last 24 hours

    58→    58→

    99→    99→        setLoadingPlan(null);
   100→   100→      }
   101→   101→    } catch {
   102→   102→      setCheckoutError(lang === "fr" ? "Erreur de connexion. Réessaie." : "Connection error. Please try again.");

   116→   116→
   117→   117→        const res = await fetch("/api/courses/paywall-status", { headers });
   118→   118→        const data = await res.json();

   120→   120→        if (data.firstName) setFirstName(data.firstName);
   121→   121→
   122→   122→        if (data.inTrial) {
   123→   123→          setInTrial(true);
   124→   124→          setTrialDaysRemaining(data.trialDaysRemaining || 0);
   125→   125→          setTrialCoursesGenerated(data.trialCoursesGenerated || 0);
   126→   126→        } else if (data.showPaywall && data.paywallReason === "trial_expired") {
   127→   127→          setTrialExpired(true);
   128→   128→        }
   129→   129→
   130→   130→        if (data.hasSubscription && data.subscriptionStatus === "active") {
   131→   131→          setIsSubscribed(true);
   132→   132→          setSubscriptionPlan(data.subscriptionPlan || "monthly");
   133→   133→          setTrialExpired(false);
   134→   134→
   135→   135→          if (data.showRenewalReminder) {
   136→   136→            setShowRenewalReminder(true);
   137→   137→            setRenewalDaysRemaining(data.renewalDaysRemaining || 0);
   138→   138→            setRenewalUrgency(data.renewalUrgency || "none");
   139→   139→            setTimeRemainingMs(data.timeRemainingMs);
   140→   140→          }
   141→   141→        }
   142→   142→
   143→   143→        if (data.inGracePeriod) {
   144→   144→          setInGracePeriod(true);
   145→   145→          setGraceDaysRemaining(data.graceDaysRemaining || 0);
   146→   146→        } else if (data.showPaywall && data.paywallReason === "grace_expired") {
   147→   147→          setGraceExpired(true);
   148→   148→        }
   149→   149→      } catch {
   150→   150→        // silently fail
   151→   151→      }
   152→   152→    };
   153→   153→    checkStatus();
   154→   154→  }, [userId]);
   155→   155→
   156→   156→  // Countdown timer for last 24 hours
   157→   157→  useEffect(() => {
   158→   158→    if (renewalUrgency !== "last24hours" || !timeRemainingMs) {
   159→   159→      if (countdownIntervalRef.current) {
   160→   160→        clearInterval(countdownIntervalRef.current);
   161→   161→        countdownIntervalRef.current = null;
   162→   162→      }
   163→   163→      return;
   164→   164→    }
   165→   165→
   166→   166→    const updateCountdown = () => {
   167→   167→      const now = Date.now();
   168→   168→      const remaining = Math.max(0, timeRemainingMs - (now - ((window as unknown as Record<string, number>).__countdownStart || now)));
   169→   169→
   170→   170→      if (remaining <= 0) {
   171→   171→        if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   172→   172→        setCountdown({ hours: 0, minutes: 0, seconds: 0 });
   173→   173→        return;
   174→   174→      }
   175→   175→
   176→   176→      const totalSeconds = Math.floor(remaining / 1000);
   177→   177→      const hours = Math.floor(totalSeconds / 3600);
   178→   178→      const minutes = Math.floor((totalSeconds % 3600) / 60);
   179→   179→      const seconds = totalSeconds % 60;
   180→   180→      setCountdown({ hours, minutes, seconds });
   181→   181→    };
   182→   182→
   183→   183→    (window as unknown as Record<string, number>).__countdownStart = Date.now();
   184→   184→
   185→   185→    const fetchFreshTime = async () => {
   186→   186→      try {
   187→   187→        const headers: Record<string, string> = {};
   188→   188→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   189→   189→        const res = await fetch("/api/courses/paywall-status", { headers });
   190→   190→        const data = await res.json();
   191→   191→        if (data.timeRemainingMs) {
   192→   192→          setTimeRemainingMs(data.timeRemainingMs);
   193→   193→          (window as unknown as Record<string, number>).__countdownStart = Date.now();
   194→   194→        }
   195→   195→      } catch { /* ignore */ }
   196→   196→    };
   197→   197→
   198→   198→    updateCountdown();
   199→   199→    countdownIntervalRef.current = setInterval(updateCountdown, 1000);

   201→   201→
   202→   202→    return () => {
   203→   203→      if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   204→   204→      clearInterval(refreshInterval);
   205→   205→    };
   206→   206→  }, [renewalUrgency, timeRemainingMs, userId]);
   207→   207→
   208→   208→  // Get renewal reminder message
   209→   209→  const getRenewalMessage = useMemo(() => {
   210→   210→    const name = firstName || (lang === "fr" ? "Bonjour" : "Hey");
   211→   211→    const isAnnual = subscriptionPlan === "annual";
   212→   212→
   213→   213→    if (renewalUrgency === "last24hours") {
   214→   214→      const key = isAnnual ? "renewalAnnual24hCountdown" : "renewalMonthly24hCountdown";
   215→   215→      return tx.offers[key]
   216→   216→        .replace("{name}", name)
   217→   217→        .replace("{hours}", String(countdown.hours))
   218→   218→        .replace("{minutes}", String(countdown.minutes).padStart(2, "0"))
   219→   219→        .replace("{seconds}", String(countdown.seconds).padStart(2, "0"));
   220→   220→    }
   221→   221→
   222→   222→    const keyMap: Record<string, string> = isAnnual
   223→   223→      ? {
   224→   224→          "1month": "renewalAnnual1Month",
   225→   225→          "2weeks": "renewalAnnual2Weeks",
   226→   226→          "1week": "renewalAnnual1Week",
   227→   227→          "3days": "renewalAnnual3Days",
   228→   228→          "24hours": "renewalAnnual24Hours",
   229→   229→        }
   230→   230→      : {
   231→   231→          "1week": "renewalMonthly1Week",
   232→   232→          "3days": "renewalMonthly3Days",
   233→   233→          "24hours": "renewalMonthly24Hours",
   234→   234→        };
   235→   235→
   236→   236→    const key = keyMap[renewalUrgency];
   237→   237→    if (key && (tx.offers as Record<string, string>)[key]) {
   238→   238→      return (tx.offers as Record<string, string>)[key].replace("{name}", name);
   239→   239→    }
   240→   240→
   241→   241→    return null;
   242→   242→  }, [renewalUrgency, firstName, subscriptionPlan, lang, countdown, tx.offers]);
   243→   243→
   244→   244→  const getUrgencyColor = () => {

   246→   246→      return "bg-red-500/10 border-red-500/30 text-red-200";
   247→   247→    }
   248→   248→    return "bg-amber-500/10 border-amber-500/30 text-amber-200";
   249→   249→  };
   250→   250→
   251→   251→  // Check URL params for payment success
   252→   252→  useEffect(() => {
   253→   253→    const params = new URLSearchParams(window.location.search);
   254→   254→    const paymentStatus = params.get("payment");
   255→   255→    const planParam = params.get("plan");
   256→   256→
   257→   257→    if (paymentStatus === "success" && planParam) {
   258→   258→      window.history.replaceState({}, "", "/");
   259→   259→      // Refresh status to show subscription is active
   260→   260→      const checkAfterCheckout = async () => {
   261→   261→        try {
   262→   262→          const headers: Record<string, string> = {};
   263→   263→          if (userId) headers["Authorization"] = `Bearer ${userId}`;
   264→   264→          const res = await fetch("/api/courses/paywall-status", { headers });
   265→   265→          const data = await res.json();
   266→   266→          if (data.hasSubscription && data.subscriptionStatus === "active") {
   267→   267→            setIsSubscribed(true);
   268→   268→            setSubscriptionPlan(planParam);
   269→   269→            setTrialExpired(false);
   270→   270→            setInGracePeriod(false);
   271→   271→            setGraceExpired(false);
   272→   272→            setShowRenewalReminder(false);
   273→   273→          }
   274→   274→        } catch { /* silent */ }
   275→   275→      };
   276→   276→      setTimeout(checkAfterCheckout, 2000);
   277→   277→    } else if (paymentStatus === "cancelled") {
   278→   278→      window.history.replaceState({}, "", "/");
   279→   279→      // Use setTimeout to avoid calling setState directly in effect
   280→   280→      setTimeout(() => {
   281→   281→        setCheckoutError(lang === "fr" ? "Paiement annulé. Tu peux réessayer." : "Payment cancelled. You can try again.");
   282→   282→      }, 0);
   283→   283→    }
   284→   284→  }, [userId, lang]);
   285→   285→
   286→   286→  const suffix = (n: number) => (n > 1 ? (lang === "fr" ? "s" : "s") : "");
   287→   287→
   288→   288→  const cannotRenewMessage = useMemo(() => {
   289→   289→    if (isSubscribed && !showRenewalReminder) {
   290→   290→      const name = firstName || "";
   291→   291→      return tx.offers.cannotRenewEarly.replace("{name}", name || (lang === "fr" ? "Bonjour" : "Hey"));
   292→   292→    }
   293→   293→    return null;
   294→   294→  }, [isSubscribed, showRenewalReminder, firstName, lang, tx.offers]);
   295→   295→
   296→   296→  // Button disabled logic
   297→   297→  const isButtonDisabled = (plan: string) =>
   298→   298→    loadingPlan === plan ||
   299→   299→    paymentProcessing ||
   300→   300→    (isSubscribed && !showRenewalReminder && !inGracePeriod && !graceExpired);
   301→   301→
   302→   302→  return (
   303→   303→    <>
   304→   304→    <div className="min-h-screen bg-night px-4 sm:px-6 md:px-10 pt-14 sm:pt-20 pb-4 sm:pb-6 md:pb-10 lg:pb-14 md:pt-24">
   305→   305→      {/* Background glow */}
   306→   306→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   307→   307→        <div className="absolute top-1/3 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[800px] h-[800px] bg-mauve/5 rounded-full blur-[150px]" />
   308→   308→      </div>
   309→   309→
   310→   310→      <div className="relative z-10 max-w-6xl mx-auto">
   311→   311→        {/* ===== HEADER ===== */}
   312→   312→        <div className="text-center mb-12">
   313→   313→          <h1 className="text-4xl sm:text-5xl md:text-6xl font-extrabold mb-4">
   314→   314→            <span className="gradient-text">{tx.offers.title}</span>
   315→   315→          </h1>
   316→   316→          <p className="text-lg sm:text-xl text-muted-foreground max-w-2xl mx-auto">
   317→   317→            {tx.offers.subtitle}
   318→   318→          </p>
   319→   319→        </div>
   320→   320→
   321→   321→        {/* ===== PAYMENT SUCCESS BANNER ===== */}
   322→   322→        {paymentSuccess && (
   323→   323→          <div className="max-w-2xl mx-auto mb-8 space-y-3">
   324→   324→            <div className="flex items-start gap-3 p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   325→   325→              <Check className="w-6 h-6 text-emerald-400 flex-shrink-0 mt-0.5" />
   326→   326→              <div>
   327→   327→                <p className="text-base font-bold text-emerald-300">
   328→   328→                  {lang === "fr" ? "Paiement réussi ! 🎉" : "Payment successful! 🎉"}
   329→   329→                </p>
   330→   330→                <p className="text-sm text-emerald-400/70 mt-1">
   331→   331→                  {lang === "fr"
   332→   332→                    ? "Ton abonnement est maintenant actif. Tu as accès à tous les cours."
   333→   333→                    : "Your subscription is now active. You have access to all courses."}
   334→   334→                </p>
   335→   335→              </div>
   336→   336→            </div>
   337→   337→          </div>
   338→   338→        )}
   339→   339→
   340→   340→        {/* ===== PAYMENT PROCESSING BANNER ===== */}
   341→   341→        {paymentProcessing && !paymentSuccess && (
   342→   342→          <div className="max-w-2xl mx-auto mb-8">
   343→   343→            <div className="flex items-center gap-3 p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   344→   344→              <Loader2 className="w-5 h-5 text-mauve-light animate-spin" />
   345→   345→              <p className="text-sm text-mauve-light font-medium">
   346→   346→                {lang === "fr" ? "Confirmation du paiement en cours..." : "Confirming payment..."}
   347→   347→              </p>
   348→   348→            </div>
   349→   349→          </div>
   350→   350→        )}
   351→   351→
   352→   352→        {/* ===== STATUS BANNERS ===== */}
   353→   353→        <div className="max-w-2xl mx-auto mb-8 space-y-3">

   359→   359→                <p className="text-xs font-bold uppercase tracking-wider mb-1 opacity-70">

   361→   361→                </p>
   362→   362→                <p className="text-sm sm:text-base font-medium">{getRenewalMessage}</p>

   368→   368→              </div>
   369→   369→            </div>

   371→   371→
   372→   372→          {/* Grace period */}

   375→   375→              <Clock className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   376→   376→              <div>
   377→   377→                <p className="text-sm sm:text-base text-amber-200 font-medium">
   378→   378→                  {tx.offers.gracePeriod.replace("{days}", String(graceDaysRemaining)).replace("{suffix}", suffix(graceDaysRemaining))}

   381→   381→              </div>
   382→   382→            </div>
   383→   383→          )}
   384→   384→
   385→   385→          {/* Grace expired */}

   390→   390→            </div>
   391→   391→          )}
   392→   392→
   393→   393→          {/* Trial active */}
   394→   394→          {inTrial && (
   395→   395→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   396→   396→              <Gift className="w-5 h-5 text-mauve-light flex-shrink-0 mt-0.5" />

   404→   404→              </div>
   405→   405→            </div>
   406→   406→          )}
   407→   407→
   408→   408→          {/* Trial expired */}
   409→   409→          {trialExpired && (
