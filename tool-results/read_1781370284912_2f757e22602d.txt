   214→   214→          "2weeks": "renewalAnnual2Weeks",
   215→   215→          "1week": "renewalAnnual1Week",

   259→   259→            setInGracePeriod(false);

   310→   310→        {/* ===== PAYMENT SUCCESS BANNER ===== */}

   317→   317→                  {lang === "fr" ? "Paiement réussi ! 🎉" : "Payment successful! 🎉"}
