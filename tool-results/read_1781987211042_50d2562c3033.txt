     9→async function withRetry<T>(fn: () => Promise<T>, retries = 2): Promise<T> {
    10→  for (let attempt = 0; attempt <= retries; attempt++) {
    11→    try { return await fn(); } catch (error: unknown) {
    12→      const msg = error instanceof Error ? error.message : String(error);
    13→      if (msg.includes("429") && attempt < retries) { await sleep(2000 * (attempt + 1)); continue; }
    14→      throw error;
    15→    }
    16→  }
    17→  throw new Error("Max retries exceeded");
    18→}

    30→async function searchOnce(
    31→  zai: Awaited<ReturnType<typeof ZAI.create>>,
    32→  query: string,
    33→): Promise<string> {
    34→  try {
    35→    const results = await zai.functions.invoke("web_search", { query, num: 5 });
    36→    if (!Array.isArray(results) || results.length === 0) return "";
    37→    return results
    38→      .slice(0, 4)
    39→      .map((r: SearchResult, i: number) => `[${i + 1}] ${r.name}: ${r.snippet}`)
    40→      .join("\n");
    41→  } catch { return ""; }
    42→}

    85→function htmlToPlainText(html: string): string {
    86→  return html
    87→    .replace(/<script[^>]*>[\s\S]*?<\/script>/gi, "")
    88→    .replace(/<style[^>]*>[\s\S]*?<\/style>/gi, "")
    89→    .replace(/<!--[\s\S]*?-->/g, "").replace(/<br\s*\/?>/gi, "\n")
    90→    .replace(/<\/p>/gi, "\n\n").replace(/<[^>]*>/g, " ")
    91→    .replace(/&nbsp;/g, " ").replace(/&amp;/g, "&").replace(/&lt;/g, "<")
    92→    .replace(/&gt;/g, ">").replace(/&quot;/g, '"').replace(/&#39;/g, "'")
    93→    .replace(/\s+/g, " ").trim();
    94→}

    96→async function scrapeSourceLinks(zai: Awaited<ReturnType<typeof ZAI.create>>, sourceLinks: string[]): Promise<ScrapedPage[]> {
    97→  const results: ScrapedPage[] = [];
    98→  for (let i = 0; i < Math.min(sourceLinks.length, MAX_SOURCE_LINKS); i++) {
    99→    const url = sourceLinks[i];
   100→    if (!url?.startsWith("http")) continue;
   101→    try {
   102→      const result = await zai.functions.invoke("page_reader", { url });
   103→      const html = result.data?.html || "";
   104→      const text = htmlToPlainText(html);
   105→      const title = result.data?.title || url;
   106→      const truncatedText = text.length > 2000 ? text.slice(0, 2000) + "..." : text;
   107→      if (truncatedText.length > 50) results.push({ title, text: truncatedText, url });
   108→    } catch { console.error(`[scrape] FAIL: ${url}`); }
   109→  }
   110→  return results;
   111→}

   236→function extractOutline(text: string): OutlineResult | null {
   237→  let cleaned = text.trim();
   238→  const cb = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/);
   239→  if (cb) cleaned = cb[1].trim();
   240→
   241→  const firstBrace = cleaned.indexOf("{");
   242→  if (firstBrace === -1) return null;
   243→
   244→  let depth = 0, lastBrace = -1;
   245→  for (let i = firstBrace; i < cleaned.length; i++) {
   246→    if (cleaned[i] === "\\") continue;
   247→    if (cleaned[i] === "\"") { let j = i + 1; while (j < cleaned.length) { if (cleaned[j] === "\\") { j += 2; continue; } if (cleaned[j] === "\"") break; j++; } i = j; continue; }
   248→    if (cleaned[i] === "{") depth++;
   249→    if (cleaned[i] === "}") { depth--; lastBrace = i; if (depth === 0) break; }
   250→    if (cleaned[i] === "[") depth++;
   251→    if (cleaned[i] === "]") depth--;
   252→  }
   253→
   254→  const snippet = lastBrace > firstBrace ? cleaned.slice(firstBrace, lastBrace + 1) : cleaned.slice(firstBrace);
   255→
   256→  try {
   257→    const data = JSON.parse(snippet) as Record<string, unknown>;
   258→    if (!data.chapters || !Array.isArray(data.chapters)) return null;
   259→    const description = typeof data.description === "string" ? data.description : "";
   260→    const chapters: OutlineChapter[] = [];
   261→    for (const ch of data.chapters) {
   262→      if (!ch || typeof ch !== "object") continue;
   263→      const c = ch as Record<string, unknown>;
   264→      if (typeof c.title === "string" && c.title.trim()) {
   265→        chapters.push({
   266→          title: c.title.trim(),
   267→          goal: typeof c.goal === "string" ? c.goal.trim() : "",
   268→          keyConcepts: Array.isArray(c.keyConcepts) ? c.keyConcepts.map(String) : [],
   269→          subSections: Array.isArray(c.subSections) ? c.subSections.map(String) : [],
   270→          plannedAnalogy: typeof c.plannedAnalogy === "string" ? c.plannedAnalogy.trim() : "",
   271→          plannedCaseStudy: typeof c.plannedCaseStudy === "string" ? c.plannedCaseStudy.trim() : "",
   272→          plannedExample: typeof c.plannedExample === "string" ? c.plannedExample.trim() : "",
   273→          mythToBust: typeof c.mythToBust === "string" ? c.mythToBust.trim() : "",
   274→          reflectionQuestion: typeof c.reflectionQuestion === "string" ? c.reflectionQuestion.trim() : "",
   275→          realAction: typeof c.realAction === "string" ? c.realAction.trim() : "",
   276→          expertNote: typeof c.expertNote === "string" ? c.expertNote.trim() : "",
   277→        });
   278→      }
   279→    }
   280→    return chapters.length > 0 ? { description, chapters } : null;
   281→  } catch {
   282→    // Recovery attempt
   283→    try {
   284→      const fixed = snippet.replace(/,\s*([}\]])/g, "$1").replace(/[\u201C\u201D\u2018\u2019]/g, "'");
   285→      const data = JSON.parse(fixed) as Record<string, unknown>;
   286→      if (!data.chapters || !Array.isArray(data.chapters)) return null;
   287→      const description = typeof data.description === "string" ? data.description : "";
   288→      const chapters: OutlineChapter[] = [];
   289→      for (const ch of data.chapters) {
   290→        if (!ch || typeof ch !== "object") continue;
   291→        const c = ch as Record<string, unknown>;
   292→        if (typeof c.title === "string" && c.title.trim()) {
   293→          chapters.push({
   294→            title: c.title.trim(), goal: typeof c.goal === "string" ? c.goal.trim() : "",
   295→            keyConcepts: Array.isArray(c.keyConcepts) ? c.keyConcepts.map(String) : [],
   296→            subSections: Array.isArray(c.subSections) ? c.subSections.map(String) : [],
   297→            plannedAnalogy: typeof c.plannedAnalogy === "string" ? c.plannedAnalogy.trim() : "",
   298→            plannedCaseStudy: typeof c.plannedCaseStudy === "string" ? c.plannedCaseStudy.trim() : "",
   299→            plannedExample: typeof c.plannedExample === "string" ? c.plannedExample.trim() : "",
   300→            mythToBust: typeof c.mythToBust === "string" ? c.mythToBust.trim() : "",
   301→            reflectionQuestion: typeof c.reflectionQuestion === "string" ? c.reflectionQuestion.trim() : "",
   302→            realAction: typeof c.realAction === "string" ? c.realAction.trim() : "",
   303→          });
   304→        }
   305→      }
   306→      return chapters.length > 0 ? { description, chapters } : null;
   307→    } catch { return null; }
   308→  }
   309→}

   463→function extractChapter(text: string): { title: string; content: string; summary: string } | null {
   464→  let cleaned = text.trim();
   465→  const cb = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/);
   466→  if (cb) cleaned = cb[1].trim();
   467→
   468→  const firstBrace = cleaned.indexOf("{");
   469→  if (firstBrace === -1) return null;
   470→
   471→  let depth = 0, lastBrace = -1;
   472→  for (let i = firstBrace; i < cleaned.length; i++) {
   473→    if (cleaned[i] === "\\") continue;
   474→    if (cleaned[i] === "\"") { let j = i + 1; while (j < cleaned.length) { if (cleaned[j] === "\\") { j += 2; continue; } if (cleaned[j] === "\"") break; j++; } i = j; continue; }
   475→    if (cleaned[i] === "{") depth++;
   476→    if (cleaned[i] === "}") { depth--; lastBrace = i; if (depth === 0) break; }
   477→    if (cleaned[i] === "[") depth++;
   478→    if (cleaned[i] === "]") depth--;
   479→  }
   480→
   481→  const snippet = lastBrace > firstBrace ? cleaned.slice(firstBrace, lastBrace + 1) : cleaned.slice(firstBrace);
   482→
   483→  const tryParse = (s: string) => {
   484→    try {
   485→      const data = JSON.parse(s) as Record<string, unknown>;
   486→      if (typeof data.title === "string" && data.title.trim() && typeof data.content === "string" && data.content.trim()) {
   487→        return { title: data.title.trim(), content: data.content.trim(), summary: typeof data.summary === "string" ? data.summary.trim() : "" };
   488→      }
   489→    } catch { /* continue */ }
   490→    return null;
   491→  };
   492→
   493→  return tryParse(snippet) || tryParse(snippet.replace(/,\s*([}\]])/g, "$1").replace(/[\u201C\u201D\u2018\u2019]/g, "'")) || null;
   494→}

   497→   STEP 3: QUALITY VALIDATION
   498→   ═══════════════════════════════════════════════════════════════════════════ */

   500→function validateChapterQuality(content: string, level: number): { passed: boolean; wordCount: number; headingCount: number; issues: string[] } {
   501→  const issues: string[] = [];
   502→  const wordCount = content.split(/\s+/).filter(Boolean).length;
   503→  const headingCount = (content.match(/^##\s+.+$/gm) || []).length;
   504→  const minWords = level >= 2 ? 500 : 400;
   505→
   506→  if (wordCount < minWords) issues.push(`Content too short (${wordCount} words, minimum ${minWords})`);
   507→  if (headingCount < 4) issues.push(`Too few sections (${headingCount}, minimum 4)`);
   508→
   509→  return { passed: issues.length === 0, wordCount, headingCount, issues };
   510→}

   540→function extractFallbackCourse(text: string): { description: string; chapters: Array<{ title: string; content: string; summary: string }> } | null {
   541→  let cleaned = text.trim();

   551→    if (cleaned[i] === "{") depth++;
   552→    if (cleaned[i] === "}") { depth--; lastBrace = i; if (depth === 0) break; }

   582→export async function POST(request: NextRequest) {
   583→  try {
   584→    const { title, sourceLinks = [], level = 0, courseLang = "fr", userId } = await request.json();
   585→
   586→    if (!title || typeof title !== "string" || title.trim().length === 0) {
   587→      return NextResponse.json({ error: "Title is required" }, { status: 400 });
   588→    }
   589→
   590→    // ── Free limit ──
   591→    if (userId) {
   592→      const [user, existingCourses] = await Promise.all([
   593→        db.user.findUnique({ where: { id: userId }, select: { subscriptionStatus: true } }),
   594→        db.course.count({ where: { userId } }),
   595→      ]);
   596→      if (user?.subscriptionStatus !== "active" && existingCourses >= FREE_COURSE_LIMIT) {
   597→        return NextResponse.json({ error: "FREE_LIMIT", requiresSubscription: true }, { status: 403 });
   598→      }
   599→    }
   600→
   601→    // ── Step 0: Deep web search + source scraping (parallel) ──
   602→    let zaiInstance: Awaited<ReturnType<typeof ZAI.create>> | null = null;
   603→    let webContext = "";
   604→    let scrapedPages: ScrapedPage[] = [];
   605→
   606→    try {
   607→      zaiInstance = await ZAI.create();
   608→      const [searchResults, scraped] = await Promise.all([
   609→        deepSearch(zaiInstance, title, courseLang, level),
   610→        sourceLinks.length > 0 ? scrapeSourceLinks(zaiInstance, sourceLinks) : Promise.resolve([]),
   611→      ]);
   612→      webContext = searchResults;
   613→      scrapedPages = scraped;
   614→    } catch {
   615→      console.log("[generate] z-ai SDK unavailable, skipping search/scraping");
   616→    }
   617→
   618→    const sourceContext = buildSourceContext(scrapedPages);
   619→    console.log(`[generate] Starting generation for "${title}" level=${level} (web: ${webContext.length > 0 ? "yes" : "no"}, sources: ${scrapedPages.length})`);
   620→
   621→    // ── Step 1: Generate outline ──
   622→    let outline = await generateOutline(title, courseLang, level, webContext, sourceContext);
   623→
   624→    if (!outline || outline.chapters.length < 3) {
   625→      console.log("[generate] Outline failed, trying single-call fallback...");
   626→      const fallbackResult = await generateSingleCall(title, courseLang, level, webContext, sourceContext);
   627→      if (!fallbackResult || fallbackResult.chapters.length === 0) {
   628→        return NextResponse.json({ error: "L'IA n'a pas pu générer un cours valide. Réessaie." }, { status: 500 });
   629→      }
   630→      const course = await saveCourse(title, level, userId, sourceLinks, fallbackResult.description, fallbackResult.chapters, scrapedPages.length);
   631→      return NextResponse.json(buildResponse(course, sourceLinks, scrapedPages.length));
   632→    }
   633→
   634→    console.log(`[outline] ${outline.chapters.length} chapters planned`);
   635→
   636→    // ── Step 2: Generate each chapter individually ──
   637→    const generatedChapters: Array<{ title: string; content: string; summary: string }> = [];
   638→
   639→    for (let i = 0; i < outline.chapters.length; i++) {
   640→      const ch = outline.chapters[i];
   641→      let chapter = await generateChapter(title, courseLang, level, i, outline.chapters.length, ch, webContext, sourceContext);
   642→
   643→      if (!chapter) {
   644→        console.log(`[chapter-${i + 1}] Failed, retrying without research context...`);
   645→        chapter = await generateChapter(title, courseLang, level, i, outline.chapters.length, ch, "", "");
   646→      }
   647→
   648→      if (chapter) {
   649→        const quality = validateChapterQuality(chapter.content, level);
   650→        if (!quality.passed) console.log(`[chapter-${i + 1}] Quality issues: ${quality.issues.join(", ")} (${quality.wordCount} words, ${quality.headingCount} headings)`);
   651→        else console.log(`[chapter-${i + 1}] Quality OK (${quality.wordCount} words, ${quality.headingCount} headings)`);
   652→        generatedChapters.push(chapter);
   653→      } else {
   654→        console.warn(`[chapter-${i + 1}] Completely failed, skipping`);
   655→      }
   656→    }
   657→
   658→    if (generatedChapters.length === 0) {
   659→      console.log("[generate] All chapters failed, trying single-call fallback...");
   660→      const fallbackResult = await generateSingleCall(title, courseLang, level, webContext, sourceContext);
   661→      if (!fallbackResult || fallbackResult.chapters.length === 0) {
   662→        return NextResponse.json({ error: "L'IA n'a pas pu générer un cours valide. Réessaie." }, { status: 500 });
   663→      }
   664→      const course = await saveCourse(title, level, userId, sourceLinks, fallbackResult.description, fallbackResult.chapters, scrapedPages.length);
   665→      return NextResponse.json(buildResponse(course, sourceLinks, scrapedPages.length));
   666→    }
   667→
   668→    console.log(`[generate] Generated ${generatedChapters.length}/${outline.chapters.length} chapters successfully`);
   669→
   670→    // ── Step 3: Save ──
   671→    const course = await saveCourse(title, level, userId, sourceLinks, outline.description, generatedChapters, scrapedPages.length);
   672→    return NextResponse.json(buildResponse(course, sourceLinks, scrapedPages.length));
   673→  } catch (error: unknown) {
   674→    console.error("Course generation error:", error);
   675→    return NextResponse.json({ error: error instanceof Error ? error.message : "Failed to generate course" }, { status: 500 });
   676→  }
   677→}
