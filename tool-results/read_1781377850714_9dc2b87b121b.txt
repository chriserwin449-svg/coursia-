     1→"use client";
     2→
     3→import {
     4→  Check,
     5→  Crown,
     6→  Zap,
     7→  AlertTriangle,
     8→  Loader2,
     9→  Clock,
    10→  ShieldAlert,
    11→  Gift,
    12→  Lock,
    13→} from "lucide-react";
    14→import { useAppStore } from "@/lib/store";
    15→import { t } from "@/lib/i18n";
    16→import { useState, useMemo, useEffect, useCallback, useRef } from "react";
    17→// PayPal buttons removed — using custom CTA buttons with redirect flow
    18→
    19→export default function OffersPage() {
    20→  const lang = useAppStore((s) => s.lang);
    21→  const userId = useAppStore((s) => s.userId);
    22→  const isAuthenticated = useAppStore((s) => s.isAuthenticated);
    23→  const setView = useAppStore((s) => s.setView);
    24→  const tx = t(lang);
    25→
    26→  const [trialExpired, setTrialExpired] = useState(false);
    27→  const [inTrial, setInTrial] = useState(false);
    28→  const [trialDaysRemaining, setTrialDaysRemaining] = useState(0);
    29→  const [trialCoursesGenerated, setTrialCoursesGenerated] = useState(0);
    30→  const [trialCoursesMax] = useState(3);
    31→  const [isSubscribed, setIsSubscribed] = useState(false);
    32→  const [subscriptionPlan, setSubscriptionPlan] = useState<string>("");
    33→  const [inGracePeriod, setInGracePeriod] = useState(false);
    34→  const [graceDaysRemaining, setGraceDaysRemaining] = useState(0);
    35→  const [graceExpired, setGraceExpired] = useState(false);
    36→  const [showRenewalReminder, setShowRenewalReminder] = useState(false);
    37→  const [renewalDaysRemaining, setRenewalDaysRemaining] = useState(0);
    38→  const [renewalUrgency, setRenewalUrgency] = useState<string>("none");
    39→  const [timeRemainingMs, setTimeRemainingMs] = useState<number | undefined>();
    40→  const [firstName, setFirstName] = useState<string>("");
    41→  const [loadingPlan, setLoadingPlan] = useState<string | null>(null);
    42→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
    43→
    44→  // PayPal payment flow states
    45→  const [paymentRequestId, setPaymentRequestId] = useState<string | null>(null);
    46→  const [paypalOrderId, setPaypalOrderId] = useState<string | null>(null);
    47→  const [paymentSuccess, setPaymentSuccess] = useState(false);
    48→  const [paymentProcessing, setPaymentProcessing] = useState(false);
    49→
    50→  // Countdown timer for last 24 hours
    51→  const [countdown, setCountdown] = useState({ hours: 0, minutes: 0, seconds: 0 });
    52→  const countdownIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
    53→
    54→  // PayPal is always assumed configured — the checkout API will handle errors if not
    55→  const isPaypalConfigured = true;
    56→
    57→  // Handle checkout — create order and redirect to PayPal
    58→  const handleCheckout = useCallback(async (plan: string) => {
    59→    if (!isAuthenticated || !userId) {
    60→      setView("auth");
    61→      return;
    62→    }
    63→    if (loadingPlan) return;
    64→
    65→    setLoadingPlan(plan);
    66→    setCheckoutError(null);
    67→
    68→    try {
    69→      const res = await fetch("/api/subscription/checkout", {
    70→        method: "POST",
    71→        headers: { "Content-Type": "application/json" },
    72→        body: JSON.stringify({ plan, userId }),
    73→      });
    74→
    75→      const data = await res.json();
    76→
    77→      if (!res.ok) {
    78→        setCheckoutError(data.error || (lang === "fr" ? "Erreur lors du paiement" : "Payment failed"));
    79→        setLoadingPlan(null);
    80→        return;
    81→      }
    82→
    83→      if (data.requestId) setPaymentRequestId(data.requestId);
    84→      setPaypalOrderId(data.orderId);
    85→
    86→      // Redirect to PayPal for approval
    87→      if (data.approveUrl) {
    88→        window.location.href = data.approveUrl;
    89→      } else {
    90→        setCheckoutError(lang === "fr" ? "Lien PayPal indisponible." : "PayPal link unavailable.");
    91→        setLoadingPlan(null);
    92→      }
    93→    } catch {
    94→      setCheckoutError(lang === "fr" ? "Erreur de connexion. Réessaie." : "Connection error. Please try again.");
    95→      setLoadingPlan(null);
    96→    }
    97→  }, [isAuthenticated, userId, lang, setView, loadingPlan]);
    98→
    99→  // Check paywall & subscription status
   100→  useEffect(() => {
   101→    const checkStatus = async () => {
   102→      try {
   103→        const headers: Record<string, string> = {};
   104→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   105→
   106→        const res = await fetch("/api/courses/paywall-status", { headers });
   107→        const data = await res.json();
   108→
   109→        if (data.firstName) setFirstName(data.firstName);
   110→
   111→        if (data.inTrial) {
   112→          setInTrial(true);
   113→          setTrialDaysRemaining(data.trialDaysRemaining || 0);
   114→          setTrialCoursesGenerated(data.trialCoursesGenerated || 0);
   115→        } else if (data.showPaywall && data.paywallReason === "trial_expired") {
   116→          setTrialExpired(true);
   117→        }
   118→
   119→        if (data.hasSubscription && data.subscriptionStatus === "active") {
   120→          setIsSubscribed(true);
   121→          setSubscriptionPlan(data.subscriptionPlan || "monthly");
   122→          setTrialExpired(false);
   123→
   124→          if (data.showRenewalReminder) {
   125→            setShowRenewalReminder(true);
   126→            setRenewalDaysRemaining(data.renewalDaysRemaining || 0);
   127→            setRenewalUrgency(data.renewalUrgency || "none");
   128→            setTimeRemainingMs(data.timeRemainingMs);
   129→          }
   130→        }
   131→
   132→        if (data.inGracePeriod) {
   133→          setInGracePeriod(true);
   134→          setGraceDaysRemaining(data.graceDaysRemaining || 0);
   135→        } else if (data.showPaywall && data.paywallReason === "grace_expired") {
   136→          setGraceExpired(true);
   137→        }
   138→      } catch {
   139→        // silently fail
   140→      }
   141→    };
   142→    checkStatus();
   143→  }, [userId]);
   144→
   145→  // Countdown timer for last 24 hours
   146→  useEffect(() => {
   147→    if (renewalUrgency !== "last24hours" || !timeRemainingMs) {
   148→      if (countdownIntervalRef.current) {
   149→        clearInterval(countdownIntervalRef.current);
   150→        countdownIntervalRef.current = null;
   151→      }
   152→      return;
   153→    }
   154→
   155→    const updateCountdown = () => {
   156→      const now = Date.now();
   157→      const remaining = Math.max(0, timeRemainingMs - (now - ((window as unknown as Record<string, number>).__countdownStart || now)));
   158→
   159→      if (remaining <= 0) {
   160→        if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   161→        setCountdown({ hours: 0, minutes: 0, seconds: 0 });
   162→        return;
   163→      }
   164→
   165→      const totalSeconds = Math.floor(remaining / 1000);
   166→      const hours = Math.floor(totalSeconds / 3600);
   167→      const minutes = Math.floor((totalSeconds % 3600) / 60);
   168→      const seconds = totalSeconds % 60;
   169→      setCountdown({ hours, minutes, seconds });
   170→    };
   171→
   172→    (window as unknown as Record<string, number>).__countdownStart = Date.now();
   173→
   174→    const fetchFreshTime = async () => {
   175→      try {
   176→        const headers: Record<string, string> = {};
   177→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   178→        const res = await fetch("/api/courses/paywall-status", { headers });
   179→        const data = await res.json();
   180→        if (data.timeRemainingMs) {
   181→          setTimeRemainingMs(data.timeRemainingMs);
   182→          (window as unknown as Record<string, number>).__countdownStart = Date.now();
   183→        }
   184→      } catch { /* ignore */ }
   185→    };
   186→
   187→    updateCountdown();
   188→    countdownIntervalRef.current = setInterval(updateCountdown, 1000);
   189→    const refreshInterval = setInterval(fetchFreshTime, 30_000);
   190→
   191→    return () => {
   192→      if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   193→      clearInterval(refreshInterval);
   194→    };
   195→  }, [renewalUrgency, timeRemainingMs, userId]);
   196→
   197→  // Get renewal reminder message
   198→  const getRenewalMessage = useMemo(() => {
   199→    const name = firstName || (lang === "fr" ? "Bonjour" : "Hey");
   200→    const isAnnual = subscriptionPlan === "annual";
   201→
   202→    if (renewalUrgency === "last24hours") {
   203→      const key = isAnnual ? "renewalAnnual24hCountdown" : "renewalMonthly24hCountdown";
   204→      return tx.offers[key]
   205→        .replace("{name}", name)
   206→        .replace("{hours}", String(countdown.hours))
   207→        .replace("{minutes}", String(countdown.minutes).padStart(2, "0"))
   208→        .replace("{seconds}", String(countdown.seconds).padStart(2, "0"));
   209→    }
   210→
   211→    const keyMap: Record<string, string> = isAnnual
   212→      ? {
   213→          "1month": "renewalAnnual1Month",
   214→          "2weeks": "renewalAnnual2Weeks",
   215→          "1week": "renewalAnnual1Week",
   216→          "3days": "renewalAnnual3Days",
   217→          "24hours": "renewalAnnual24Hours",
   218→        }
   219→      : {
   220→          "1week": "renewalMonthly1Week",
   221→          "3days": "renewalMonthly3Days",
   222→          "24hours": "renewalMonthly24Hours",
   223→        };
   224→
   225→    const key = keyMap[renewalUrgency];
   226→    if (key && (tx.offers as Record<string, string>)[key]) {
   227→      return (tx.offers as Record<string, string>)[key].replace("{name}", name);
   228→    }
   229→
   230→    return null;
   231→  }, [renewalUrgency, firstName, subscriptionPlan, lang, countdown, tx.offers]);
   232→
   233→  const getUrgencyColor = () => {
   234→    if (renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days") {
   235→      return "bg-red-500/10 border-red-500/30 text-red-200";
   236→    }
   237→    return "bg-amber-500/10 border-amber-500/30 text-amber-200";
   238→  };
   239→
   240→  // Check URL params for payment success
   241→  useEffect(() => {
   242→    const params = new URLSearchParams(window.location.search);
   243→    const paymentStatus = params.get("payment");
   244→    const planParam = params.get("plan");
   245→
   246→    if (paymentStatus === "success" && planParam) {
   247→      window.history.replaceState({}, "", "/");
   248→      // Refresh status to show subscription is active
   249→      const checkAfterCheckout = async () => {
   250→        try {
   251→          const headers: Record<string, string> = {};
   252→          if (userId) headers["Authorization"] = `Bearer ${userId}`;
   253→          const res = await fetch("/api/courses/paywall-status", { headers });
   254→          const data = await res.json();
   255→          if (data.hasSubscription && data.subscriptionStatus === "active") {
   256→            setIsSubscribed(true);
   257→            setSubscriptionPlan(planParam);
   258→            setTrialExpired(false);
   259→            setInGracePeriod(false);
   260→            setGraceExpired(false);
   261→            setShowRenewalReminder(false);
   262→          }
   263→        } catch { /* silent */ }
   264→      };
   265→      setTimeout(checkAfterCheckout, 2000);
   266→    } else if (paymentStatus === "cancelled") {
   267→      window.history.replaceState({}, "", "/");
   268→      // Use setTimeout to avoid calling setState directly in effect
   269→      setTimeout(() => {
   270→        setCheckoutError(lang === "fr" ? "Paiement annulé. Tu peux réessayer." : "Payment cancelled. You can try again.");
   271→      }, 0);
   272→    }
   273→  }, [userId, lang]);
   274→
   275→  const suffix = (n: number) => (n > 1 ? (lang === "fr" ? "s" : "s") : "");
   276→
   277→  const cannotRenewMessage = useMemo(() => {
   278→    if (isSubscribed && !showRenewalReminder) {
   279→      const name = firstName || "";
   280→      return tx.offers.cannotRenewEarly.replace("{name}", name || (lang === "fr" ? "Bonjour" : "Hey"));
   281→    }
   282→    return null;
   283→  }, [isSubscribed, showRenewalReminder, firstName, lang, tx.offers]);
   284→
   285→  // Button disabled logic
   286→  const isButtonDisabled = (plan: string) =>
   287→    loadingPlan === plan ||
   288→    paymentProcessing ||
   289→    (isSubscribed && !showRenewalReminder && !inGracePeriod && !graceExpired);
   290→
   291→  return (
   292→    <>
   293→    <div className="min-h-screen bg-night px-4 sm:px-6 md:px-10 pt-14 sm:pt-20 pb-4 sm:pb-6 md:pb-10 lg:pb-14 md:pt-24">
   294→      {/* Background glow */}
   295→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   296→        <div className="absolute top-1/3 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[800px] h-[800px] bg-mauve/5 rounded-full blur-[150px]" />
   297→      </div>
   298→
   299→      <div className="relative z-10 max-w-6xl mx-auto">
   300→        {/* ===== HEADER ===== */}
   301→        <div className="text-center mb-12">
   302→          <h1 className="text-4xl sm:text-5xl md:text-6xl font-extrabold mb-4">
   303→            <span className="gradient-text">{tx.offers.title}</span>
   304→          </h1>
   305→          <p className="text-lg sm:text-xl text-muted-foreground max-w-2xl mx-auto">
   306→            {tx.offers.subtitle}
   307→          </p>
   308→        </div>
   309→
   310→        {/* ===== PAYMENT SUCCESS BANNER ===== */}
   311→        {paymentSuccess && (
   312→          <div className="max-w-2xl mx-auto mb-8 space-y-3">
   313→            <div className="flex items-start gap-3 p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   314→              <Check className="w-6 h-6 text-emerald-400 flex-shrink-0 mt-0.5" />
   315→              <div>
   316→                <p className="text-base font-bold text-emerald-300">
   317→                  {lang === "fr" ? "Paiement réussi ! 🎉" : "Payment successful! 🎉"}
   318→                </p>
   319→                <p className="text-sm text-emerald-400/70 mt-1">
   320→                  {lang === "fr"
   321→                    ? "Ton abonnement est maintenant actif. Tu as accès à tous les cours."
   322→                    : "Your subscription is now active. You have access to all courses."}
   323→                </p>
   324→              </div>
   325→            </div>
   326→          </div>
   327→        )}
   328→
   329→        {/* ===== PAYMENT PROCESSING BANNER ===== */}
   330→        {paymentProcessing && !paymentSuccess && (
   331→          <div className="max-w-2xl mx-auto mb-8">
   332→            <div className="flex items-center gap-3 p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   333→              <Loader2 className="w-5 h-5 text-mauve-light animate-spin" />
   334→              <p className="text-sm text-mauve-light font-medium">
   335→                {lang === "fr" ? "Confirmation du paiement en cours..." : "Confirming payment..."}
   336→              </p>
   337→            </div>
   338→          </div>
   339→        )}
   340→
   341→        {/* ===== STATUS BANNERS ===== */}
   342→        <div className="max-w-2xl mx-auto mb-8 space-y-3">
   343→          {/* Personalized renewal reminder */}
   344→          {showRenewalReminder && getRenewalMessage && (
   345→            <div className={`flex items-start gap-3 p-4 sm:p-5 rounded-2xl border animate-fade-in ${getUrgencyColor()}`}>
   346→              <Clock className={`w-5 h-5 flex-shrink-0 mt-0.5 ${renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days" ? "text-red-400" : "text-amber-400"}`} />
   347→              <div>
   348→                <p className="text-xs font-bold uppercase tracking-wider mb-1 opacity-70">
   349→                  {tx.offers.subscriptionEndingSoon}
   350→                </p>
   351→                <p className="text-sm sm:text-base font-medium">{getRenewalMessage}</p>
   352→                {renewalUrgency === "last24hours" && (
   353→                  <p className="text-2xl sm:text-3xl font-extrabold mt-2 tabular-nums tracking-wide">
   354→                    {String(countdown.hours).padStart(2, "0")}:{String(countdown.minutes).padStart(2, "0")}:{String(countdown.seconds).padStart(2, "0")}
   355→                  </p>
   356→                )}
   357→              </div>
   358→            </div>
   359→          )}
   360→
   361→          {/* Grace period */}
   362→          {inGracePeriod && (
   363→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   364→              <Clock className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   365→              <div>
   366→                <p className="text-sm sm:text-base text-amber-200 font-medium">
   367→                  {tx.offers.gracePeriod.replace("{days}", String(graceDaysRemaining)).replace("{suffix}", suffix(graceDaysRemaining))}
   368→                </p>
   369→                <p className="text-xs text-amber-300/70 mt-1">{tx.offers.graceReadonly}</p>
   370→              </div>
   371→            </div>
   372→          )}
   373→
   374→          {/* Grace expired */}
   375→          {graceExpired && (
   376→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   377→              <ShieldAlert className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   378→              <p className="text-sm sm:text-base text-red-200 font-medium">{tx.offers.gracePeriodExpired}</p>
   379→            </div>
   380→          )}
   381→
   382→          {/* Trial active */}
   383→          {inTrial && (
   384→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   385→              <Gift className="w-5 h-5 text-mauve-light flex-shrink-0 mt-0.5" />
   386→              <div>
   387→                <p className="text-sm sm:text-base text-mauve-light font-medium">
   388→                  {tx.offers.daysRemaining.replace("{days}", String(trialDaysRemaining)).replace("{suffix}", suffix(trialDaysRemaining))}
   389→                </p>
   390→                <p className="text-xs text-mauve-light/70 mt-1">
   391→                  {tx.offers.coursesRemaining.replace("{count}", String(trialCoursesMax - trialCoursesGenerated)).replace("{suffix}", suffix(trialCoursesMax - trialCoursesGenerated))}
   392→                </p>
   393→              </div>
   394→            </div>
   395→          )}
   396→
   397→          {/* Trial expired */}
   398→          {trialExpired && (
   399→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   400→              <AlertTriangle className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   401→              <p className="text-sm sm:text-base text-amber-200 font-medium">{tx.offers.trialExpired}</p>
   402→            </div>
   403→          )}
   404→
   405→          {/* Already subscribed (not ending) */}
   406→          {isSubscribed && !showRenewalReminder && (
   407→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   408→              <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   409→              <div>
   410→                <p className="text-sm sm:text-base text-emerald-200 font-medium">{tx.offers.subscribed}</p>
   411→                {cannotRenewMessage && (
   412→                  <p className="text-xs text-emerald-300/70 mt-1">{cannotRenewMessage}</p>
   413→                )}
   414→              </div>
   415→            </div>
   416→          )}
   417→
   418→          {/* Checkout error */}
   419→          {checkoutError && (
   420→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   421→              <AlertTriangle className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   422→              <p className="text-sm sm:text-base text-red-200 font-medium">{checkoutError}</p>
   423→            </div>
   424→          )}
   425→        </div>
   426→
   427→        {/* ===== PRICING CARDS ===== */}
   428→        {!paymentSuccess && (
   429→          <div
   430→            className={`grid gap-6 lg:gap-8 items-start mb-20 grid-cols-1 md:grid-cols-2 max-w-4xl mx-auto`}
   431→          >
   432→            {/* MONTHLY PLAN */}
   433→            <div className="pricing-card-float monthly-card-glow glass rounded-3xl p-5 sm:p-8 flex flex-col hover:border-mauve/30 transition-all duration-300">
   434→              <div className="mb-6">
   435→                <div className="flex items-center gap-3 mb-2">
   436→                  <div className="w-11 h-11 sm:w-12 sm:h-12 rounded-2xl bg-mauve/10 flex items-center justify-center">
   437→                    <Zap className="w-6 h-6 text-mauve-light" />
   438→                  </div>
   439→                  <h3 className="text-lg sm:text-xl font-bold">{tx.landing.pricing.monthly.name}</h3>
   440→                </div>
   441→                <p className="text-muted-foreground text-sm">{tx.landing.pricing.monthly.desc}</p>
   442→              </div>
   443→              <div className="mb-6">
   444→                <span className="text-3xl sm:text-4xl font-extrabold">{tx.landing.pricing.monthly.price}</span>
   445→                <span className="text-lg text-muted-foreground">{tx.landing.pricing.monthly.period}</span>
   446→              </div>
   447→              <ul className="flex-1 space-y-2 sm:space-y-3 mb-8">
   448→                {tx.landing.pricing.monthly.features.map((f) => (
   449→                  <li key={f} className="flex items-start gap-3">
   450→                    <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   451→                    <span className="text-sm text-muted-foreground">{f}</span>
   452→                  </li>
   453→                ))}
   454→              </ul>
   455→
   456→              {/* CTA Button */}
   457→              {isButtonDisabled("monthly") ? (
   458→                <button
   459→                  disabled
   460→                  className="w-full py-3.5 sm:py-4 rounded-full bg-gradient-to-r from-mauve to-mauve-dark text-white font-bold opacity-50 cursor-not-allowed flex items-center justify-center gap-2"
   461→                >
   462→                  {isSubscribed && !showRenewalReminder && !inGracePeriod && !graceExpired
   463→                    ? lang === "fr" ? "Plan Actuel" : "Current Plan"
   464→                    : <>
   465→                        <Loader2 className="w-5 h-5 animate-spin" />
   466→                        {lang === "fr" ? "Chargement..." : "Loading..."}
   467→                      </>}
   468→                </button>
   469→              ) : (
   470→                <button
