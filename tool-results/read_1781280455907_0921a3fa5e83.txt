     1→"use client";
     2→
     3→import {
     4→  Check,
     5→  Crown,
     6→  Zap,
     7→  AlertTriangle,
     8→  Loader2,
     9→  Clock,
    10→  ShieldAlert,
    11→  Gift,
    12→  CreditCard,
    13→  Lock,
    14→} from "lucide-react";
    15→import { useAppStore } from "@/lib/store";
    16→import { t } from "@/lib/i18n";
    17→import { useState, useMemo, useEffect, useCallback, useRef } from "react";
    18→import { PayPalButtons } from "@paypal/react-paypal-js";
    19→
    20→export default function OffersPage() {
    21→  const lang = useAppStore((s) => s.lang);
    22→  const userId = useAppStore((s) => s.userId);
    23→  const isAuthenticated = useAppStore((s) => s.isAuthenticated);
    24→  const setView = useAppStore((s) => s.setView);
    25→  const tx = t(lang);
    26→
    27→  const [trialExpired, setTrialExpired] = useState(false);
    28→  const [inTrial, setInTrial] = useState(false);
    29→  const [trialDaysRemaining, setTrialDaysRemaining] = useState(0);
    30→  const [trialCoursesGenerated, setTrialCoursesGenerated] = useState(0);
    31→  const [trialCoursesMax] = useState(3);
    32→  const [isSubscribed, setIsSubscribed] = useState(false);
    33→  const [subscriptionPlan, setSubscriptionPlan] = useState<string>("");
    34→  const [inGracePeriod, setInGracePeriod] = useState(false);
    35→  const [graceDaysRemaining, setGraceDaysRemaining] = useState(0);
    36→  const [graceExpired, setGraceExpired] = useState(false);
    37→  const [showRenewalReminder, setShowRenewalReminder] = useState(false);
    38→  const [renewalDaysRemaining, setRenewalDaysRemaining] = useState(0);
    39→  const [renewalUrgency, setRenewalUrgency] = useState<string>("none");
    40→  const [timeRemainingMs, setTimeRemainingMs] = useState<number | undefined>();
    41→  const [firstName, setFirstName] = useState<string>("");
    42→  const [loadingPlan, setLoadingPlan] = useState<string | null>(null);
    43→  const [checkoutError, setCheckoutError] = useState<string | null>(null);
    44→
    45→  // PayPal payment flow states
    46→  const [paymentRequestId, setPaymentRequestId] = useState<string | null>(null);
    47→  const [paypalOrderId, setPaypalOrderId] = useState<string | null>(null);
    48→  const [paymentSuccess, setPaymentSuccess] = useState(false);
    49→  const [paymentProcessing, setPaymentProcessing] = useState(false);
    50→
    51→  // Countdown timer for last 24 hours
    52→  const [countdown, setCountdown] = useState({ hours: 0, minutes: 0, seconds: 0 });
    53→  const countdownIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
    54→
    55→  // Check if PayPal is configured
    56→  const paypalClientId = process.env.NEXT_PUBLIC_PAYPAL_CLIENT_ID || "";
    57→  const isPaypalConfigured = paypalClientId && paypalClientId !== "YOUR_PAYPAL_SANDBOX_CLIENT_ID";
    58→
    59→  // Check paywall & subscription status
    60→  useEffect(() => {
    61→    const checkStatus = async () => {
    62→      try {
    63→        const headers: Record<string, string> = {};
    64→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
    65→
    66→        const res = await fetch("/api/courses/paywall-status", { headers });
    67→        const data = await res.json();
    68→
    69→        if (data.firstName) setFirstName(data.firstName);
    70→
    71→        if (data.inTrial) {
    72→          setInTrial(true);
    73→          setTrialDaysRemaining(data.trialDaysRemaining || 0);
    74→          setTrialCoursesGenerated(data.trialCoursesGenerated || 0);
    75→        } else if (data.showPaywall && data.paywallReason === "trial_expired") {
    76→          setTrialExpired(true);
    77→        }
    78→
    79→        if (data.hasSubscription && data.subscriptionStatus === "active") {
    80→          setIsSubscribed(true);
    81→          setSubscriptionPlan(data.subscriptionPlan || "monthly");
    82→          setTrialExpired(false);
    83→
    84→          if (data.showRenewalReminder) {
    85→            setShowRenewalReminder(true);
    86→            setRenewalDaysRemaining(data.renewalDaysRemaining || 0);
    87→            setRenewalUrgency(data.renewalUrgency || "none");
    88→            setTimeRemainingMs(data.timeRemainingMs);
    89→          }
    90→        }
    91→
    92→        if (data.inGracePeriod) {
    93→          setInGracePeriod(true);
    94→          setGraceDaysRemaining(data.graceDaysRemaining || 0);
    95→        } else if (data.showPaywall && data.paywallReason === "grace_expired") {
    96→          setGraceExpired(true);
    97→        }
    98→      } catch {
    99→        // silently fail
   100→      }
   101→    };
   102→    checkStatus();
   103→  }, [userId]);
   104→
   105→  // Countdown timer for last 24 hours
   106→  useEffect(() => {
   107→    if (renewalUrgency !== "last24hours" || !timeRemainingMs) {
   108→      if (countdownIntervalRef.current) {
   109→        clearInterval(countdownIntervalRef.current);
   110→        countdownIntervalRef.current = null;
   111→      }
   112→      return;
   113→    }
   114→
   115→    const updateCountdown = () => {
   116→      const now = Date.now();
   117→      const remaining = Math.max(0, timeRemainingMs - (now - ((window as unknown as Record<string, number>).__countdownStart || now)));
   118→
   119→      if (remaining <= 0) {
   120→        if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   121→        setCountdown({ hours: 0, minutes: 0, seconds: 0 });
   122→        return;
   123→      }
   124→
   125→      const totalSeconds = Math.floor(remaining / 1000);
   126→      const hours = Math.floor(totalSeconds / 3600);
   127→      const minutes = Math.floor((totalSeconds % 3600) / 60);
   128→      const seconds = totalSeconds % 60;
   129→      setCountdown({ hours, minutes, seconds });
   130→    };
   131→
   132→    (window as unknown as Record<string, number>).__countdownStart = Date.now();
   133→
   134→    const fetchFreshTime = async () => {
   135→      try {
   136→        const headers: Record<string, string> = {};
   137→        if (userId) headers["Authorization"] = `Bearer ${userId}`;
   138→        const res = await fetch("/api/courses/paywall-status", { headers });
   139→        const data = await res.json();
   140→        if (data.timeRemainingMs) {
   141→          setTimeRemainingMs(data.timeRemainingMs);
   142→          (window as unknown as Record<string, number>).__countdownStart = Date.now();
   143→        }
   144→      } catch { /* ignore */ }
   145→    };
   146→
   147→    updateCountdown();
   148→    countdownIntervalRef.current = setInterval(updateCountdown, 1000);
   149→    const refreshInterval = setInterval(fetchFreshTime, 30_000);
   150→
   151→    return () => {
   152→      if (countdownIntervalRef.current) clearInterval(countdownIntervalRef.current);
   153→      clearInterval(refreshInterval);
   154→    };
   155→  }, [renewalUrgency, timeRemainingMs, userId]);
   156→
   157→  // Get renewal reminder message
   158→  const getRenewalMessage = useMemo(() => {
   159→    const name = firstName || (lang === "fr" ? "Bonjour" : "Hey");
   160→    const isAnnual = subscriptionPlan === "annual";
   161→
   162→    if (renewalUrgency === "last24hours") {
   163→      const key = isAnnual ? "renewalAnnual24hCountdown" : "renewalMonthly24hCountdown";
   164→      return tx.offers[key]
   165→        .replace("{name}", name)
   166→        .replace("{hours}", String(countdown.hours))
   167→        .replace("{minutes}", String(countdown.minutes).padStart(2, "0"))
   168→        .replace("{seconds}", String(countdown.seconds).padStart(2, "0"));
   169→    }
   170→
   171→    const keyMap: Record<string, string> = isAnnual
   172→      ? {
   173→          "1month": "renewalAnnual1Month",
   174→          "2weeks": "renewalAnnual2Weeks",
   175→          "1week": "renewalAnnual1Week",
   176→          "3days": "renewalAnnual3Days",
   177→          "24hours": "renewalAnnual24Hours",
   178→        }
   179→      : {
   180→          "1week": "renewalMonthly1Week",
   181→          "3days": "renewalMonthly3Days",
   182→          "24hours": "renewalMonthly24Hours",
   183→        };
   184→
   185→    const key = keyMap[renewalUrgency];
   186→    if (key && (tx.offers as Record<string, string>)[key]) {
   187→      return (tx.offers as Record<string, string>)[key].replace("{name}", name);
   188→    }
   189→
   190→    return null;
   191→  }, [renewalUrgency, firstName, subscriptionPlan, lang, countdown, tx.offers]);
   192→
   193→  const getUrgencyColor = () => {
   194→    if (renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days") {
   195→      return "bg-red-500/10 border-red-500/30 text-red-200";
   196→    }
   197→    return "bg-amber-500/10 border-amber-500/30 text-amber-200";
   198→  };
   199→
   200→  // Handle PayPal onApprove — capture payment on backend
   201→  const handleApprove = useCallback(async (plan: string, orderId: string) => {
   202→    setPaymentProcessing(true);
   203→    setPaypalOrderId(orderId);
   204→    setCheckoutError(null);
   205→
   206→    try {
   207→      const res = await fetch("/api/subscription/capture", {
   208→        method: "POST",
   209→        headers: { "Content-Type": "application/json" },
   210→        body: JSON.stringify({ orderId, userId }),
   211→      });
   212→
   213→      const data = await res.json();
   214→
   215→      if (data.success) {
   216→        setPaymentSuccess(true);
   217→        setPaymentProcessing(false);
   218→
   219→        // Poll subscription status to update UI
   220→        const pollInterval = setInterval(async () => {
   221→          try {
   222→            const headers: Record<string, string> = {};
   223→            if (userId) headers["Authorization"] = `Bearer ${userId}`;
   224→            const statusRes = await fetch("/api/courses/paywall-status", { headers });
   225→            const statusData = await statusRes.json();
   226→            if (statusData.hasSubscription && statusData.subscriptionStatus === "active") {
   227→              clearInterval(pollInterval);
   228→              setIsSubscribed(true);
   229→              setSubscriptionPlan(plan);
   230→              setTrialExpired(false);
   231→              setInGracePeriod(false);
   232→              setGraceExpired(false);
   233→              setShowRenewalReminder(false);
   234→            }
   235→          } catch { /* keep polling */ }
   236→        }, 5_000);
   237→
   238→        setTimeout(() => clearInterval(pollInterval), 120_000);
   239→      } else {
   240→        setCheckoutError(data.error || (lang === "fr" ? "Échec du paiement" : "Payment failed"));
   241→        setPaymentProcessing(false);
   242→      }
   243→    } catch {
   244→      setCheckoutError(lang === "fr" ? "Erreur de connexion" : "Connection error");
   245→      setPaymentProcessing(false);
   246→    }
   247→  }, [userId, lang]);
   248→
   249→  // Create PayPal order on backend
   250→  const createOrder = useCallback(async (plan: string) => {
   251→    if (!isAuthenticated || !userId) {
   252→      setView("auth");
   253→      return "";
   254→    }
   255→
   256→    setLoadingPlan(plan);
   257→    setCheckoutError(null);
   258→
   259→    try {
   260→      const res = await fetch("/api/subscription/checkout", {
   261→        method: "POST",
   262→        headers: { "Content-Type": "application/json" },
   263→        body: JSON.stringify({ plan, userId }),
   264→      });
   265→
   266→      const data = await res.json();
   267→
   268→      if (!res.ok) {
   269→        setCheckoutError(data.error || (lang === "fr" ? "Erreur lors du paiement" : "Payment failed"));
   270→        setLoadingPlan(null);
   271→        return "";
   272→      }
   273→
   274→      if (data.requestId) setPaymentRequestId(data.requestId);
   275→      setLoadingPlan(null);
   276→      return data.orderId; // Return PayPal order ID for the SDK
   277→    } catch {
   278→      setCheckoutError(lang === "fr" ? "Erreur de connexion. Réessaie." : "Connection error. Please try again.");
   279→      setLoadingPlan(null);
   280→      return "";
   281→    }
   282→  }, [isAuthenticated, userId, lang, setView]);
   283→
   284→  // Check URL params for payment success
   285→  useEffect(() => {
   286→    const params = new URLSearchParams(window.location.search);
   287→    const paymentStatus = params.get("payment");
   288→    const planParam = params.get("plan");
   289→
   290→    if (paymentStatus === "success" && planParam) {
   291→      window.history.replaceState({}, "", "/");
   292→      // Refresh status to show subscription is active
   293→      const checkAfterCheckout = async () => {
   294→        try {
   295→          const headers: Record<string, string> = {};
   296→          if (userId) headers["Authorization"] = `Bearer ${userId}`;
   297→          const res = await fetch("/api/courses/paywall-status", { headers });
   298→          const data = await res.json();
   299→          if (data.hasSubscription && data.subscriptionStatus === "active") {
   300→            setIsSubscribed(true);
   301→            setSubscriptionPlan(planParam);
   302→            setTrialExpired(false);
   303→            setInGracePeriod(false);
   304→            setGraceExpired(false);
   305→            setShowRenewalReminder(false);
   306→          }
   307→        } catch { /* silent */ }
   308→      };
   309→      setTimeout(checkAfterCheckout, 2000);
   310→    } else if (paymentStatus === "cancelled") {
   311→      window.history.replaceState({}, "", "/");
   312→      // Use setTimeout to avoid calling setState directly in effect
   313→      setTimeout(() => {
   314→        setCheckoutError(lang === "fr" ? "Paiement annulé. Tu peux réessayer." : "Payment cancelled. You can try again.");
   315→      }, 0);
   316→    }
   317→  }, [userId, lang]);
   318→
   319→  const suffix = (n: number) => (n > 1 ? (lang === "fr" ? "s" : "s") : "");
   320→
   321→  const cannotRenewMessage = useMemo(() => {
   322→    if (isSubscribed && !showRenewalReminder) {
   323→      const name = firstName || "";
   324→      return tx.offers.cannotRenewEarly.replace("{name}", name || (lang === "fr" ? "Bonjour" : "Hey"));
   325→    }
   326→    return null;
   327→  }, [isSubscribed, showRenewalReminder, firstName, lang, tx.offers]);
   328→
   329→  // Button disabled logic
   330→  const isButtonDisabled = (plan: string) =>
   331→    loadingPlan === plan ||
   332→    paymentProcessing ||
   333→    (isSubscribed && !showRenewalReminder && !inGracePeriod && !graceExpired);
   334→
   335→  // PayPal button styling
   336→  const paypalButtonStyle = {
   337→    layout: "vertical" as const,
   338→    color: "gold" as const,
   339→    shape: "pill" as const,
   340→    label: "pay" as const,
   341→    height: 45,
   342→  };
   343→
   344→  return (
   345→    <>
   346→    <div className="min-h-screen bg-night px-4 sm:px-6 md:px-10 pt-14 sm:pt-20 pb-4 sm:pb-6 md:pb-10 lg:pb-14 md:pt-24">
   347→      {/* Background glow */}
   348→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   349→        <div className="absolute top-1/3 left-1/2 -translate-x-1/2 -translate-y-1/2 w-[800px] h-[800px] bg-mauve/5 rounded-full blur-[150px]" />
   350→      </div>
   351→
   352→      <div className="relative z-10 max-w-6xl mx-auto">
   353→        {/* ===== HEADER ===== */}
   354→        <div className="text-center mb-12">
   355→          <h1 className="text-4xl sm:text-5xl md:text-6xl font-extrabold mb-4">
   356→            <span className="gradient-text">{tx.offers.title}</span>
   357→          </h1>
   358→          <p className="text-lg sm:text-xl text-muted-foreground max-w-2xl mx-auto">
   359→            {tx.offers.subtitle}
   360→          </p>
   361→        </div>
   362→
   363→        {/* ===== PAYMENT SUCCESS BANNER ===== */}
   364→        {paymentSuccess && (
   365→          <div className="max-w-2xl mx-auto mb-8 space-y-3">
   366→            <div className="flex items-start gap-3 p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   367→              <Check className="w-6 h-6 text-emerald-400 flex-shrink-0 mt-0.5" />
   368→              <div>
   369→                <p className="text-base font-bold text-emerald-300">
   370→                  {lang === "fr" ? "Paiement réussi ! 🎉" : "Payment successful! 🎉"}
   371→                </p>
   372→                <p className="text-sm text-emerald-400/70 mt-1">
   373→                  {lang === "fr"
   374→                    ? "Ton abonnement est maintenant actif. Tu as accès à tous les cours."
   375→                    : "Your subscription is now active. You have access to all courses."}
   376→                </p>
   377→              </div>
   378→            </div>
   379→          </div>
   380→        )}
   381→
   382→        {/* ===== PAYMENT PROCESSING BANNER ===== */}
   383→        {paymentProcessing && !paymentSuccess && (
   384→          <div className="max-w-2xl mx-auto mb-8">
   385→            <div className="flex items-center gap-3 p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   386→              <Loader2 className="w-5 h-5 text-mauve-light animate-spin" />
   387→              <p className="text-sm text-mauve-light font-medium">
   388→                {lang === "fr" ? "Confirmation du paiement en cours..." : "Confirming payment..."}
   389→              </p>
   390→            </div>
   391→          </div>
   392→        )}
   393→
   394→        {/* ===== STATUS BANNERS ===== */}
   395→        <div className="max-w-2xl mx-auto mb-8 space-y-3">
   396→          {/* Personalized renewal reminder */}
   397→          {showRenewalReminder && getRenewalMessage && (
   398→            <div className={`flex items-start gap-3 p-4 sm:p-5 rounded-2xl border animate-fade-in ${getUrgencyColor()}`}>
   399→              <Clock className={`w-5 h-5 flex-shrink-0 mt-0.5 ${renewalUrgency === "last24hours" || renewalUrgency === "24hours" || renewalUrgency === "3days" ? "text-red-400" : "text-amber-400"}`} />
   400→              <div>
   401→                <p className="text-xs font-bold uppercase tracking-wider mb-1 opacity-70">
   402→                  {tx.offers.subscriptionEndingSoon}
   403→                </p>
   404→                <p className="text-sm sm:text-base font-medium">{getRenewalMessage}</p>
   405→                {renewalUrgency === "last24hours" && (
   406→                  <p className="text-2xl sm:text-3xl font-extrabold mt-2 tabular-nums tracking-wide">
   407→                    {String(countdown.hours).padStart(2, "0")}:{String(countdown.minutes).padStart(2, "0")}:{String(countdown.seconds).padStart(2, "0")}
   408→                  </p>
   409→                )}
   410→              </div>
   411→            </div>
   412→          )}
   413→
   414→          {/* Grace period */}
   415→          {inGracePeriod && (
   416→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   417→              <Clock className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   418→              <div>
   419→                <p className="text-sm sm:text-base text-amber-200 font-medium">
   420→                  {tx.offers.gracePeriod.replace("{days}", String(graceDaysRemaining)).replace("{suffix}", suffix(graceDaysRemaining))}
   421→                </p>
   422→                <p className="text-xs text-amber-300/70 mt-1">{tx.offers.graceReadonly}</p>
   423→              </div>
   424→            </div>
   425→          )}
   426→
   427→          {/* Grace expired */}
   428→          {graceExpired && (
   429→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   430→              <ShieldAlert className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   431→              <p className="text-sm sm:text-base text-red-200 font-medium">{tx.offers.gracePeriodExpired}</p>
   432→            </div>
   433→          )}
   434→
   435→          {/* Trial active */}
   436→          {inTrial && (
   437→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-mauve/10 border border-mauve/30 animate-fade-in">
   438→              <Gift className="w-5 h-5 text-mauve-light flex-shrink-0 mt-0.5" />
   439→              <div>
   440→                <p className="text-sm sm:text-base text-mauve-light font-medium">
   441→                  {tx.offers.daysRemaining.replace("{days}", String(trialDaysRemaining)).replace("{suffix}", suffix(trialDaysRemaining))}
   442→                </p>
   443→                <p className="text-xs text-mauve-light/70 mt-1">
   444→                  {tx.offers.coursesRemaining.replace("{count}", String(trialCoursesMax - trialCoursesGenerated)).replace("{suffix}", suffix(trialCoursesMax - trialCoursesGenerated))}
   445→                </p>
   446→              </div>
   447→            </div>
   448→          )}
   449→
   450→          {/* Trial expired */}
   451→          {trialExpired && (
   452→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-amber-500/10 border border-amber-500/30 animate-fade-in">
   453→              <AlertTriangle className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   454→              <p className="text-sm sm:text-base text-amber-200 font-medium">{tx.offers.trialExpired}</p>
   455→            </div>
   456→          )}
   457→
   458→          {/* Already subscribed (not ending) */}
   459→          {isSubscribed && !showRenewalReminder && (
   460→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-emerald-500/10 border border-emerald-500/30 animate-fade-in">
   461→              <Check className="w-5 h-5 text-emerald-400 flex-shrink-0 mt-0.5" />
   462→              <div>
   463→                <p className="text-sm sm:text-base text-emerald-200 font-medium">{tx.offers.subscribed}</p>
   464→                {cannotRenewMessage && (
   465→                  <p className="text-xs text-emerald-300/70 mt-1">{cannotRenewMessage}</p>
   466→                )}
   467→              </div>
   468→            </div>
   469→          )}
   470→
   471→          {/* Checkout error */}
   472→          {checkoutError && (
   473→            <div className="flex items-start gap-3 p-4 sm:p-5 rounded-2xl bg-red-500/10 border border-red-500/30 animate-fade-in">
   474→              <AlertTriangle className="w-5 h-5 text-red-400 flex-shrink-0 mt-0.5" />
   475→              <p className="text-sm sm:text-base text-red-200 font-medium">{checkoutError}</p>
   476→            </div>
   477→          )}
   478→        </div>
   479→
   480→        {/* ===== PAYPAL NOT CONFIGURED NOTICE ===== */}
   481→        {!isPaypalConfigured && (
   482→          <div className="max-w-2xl mx-auto mb-8">
   483→            <div className="flex items-start gap-3 p-4 rounded-2xl bg-amber-500/10 border border-amber-500/30">
   484→              <AlertTriangle className="w-5 h-5 text-amber-400 flex-shrink-0 mt-0.5" />
   485→              <div>
   486→                <p className="text-sm font-bold text-amber-300">
