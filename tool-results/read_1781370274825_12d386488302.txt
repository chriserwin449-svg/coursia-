     1→"use client";
     2→
     3→import { useState, useEffect, useCallback, useRef, useMemo } from "react";
     4→import {
     5→  ChevronRight,
     6→  ChevronLeft,
     7→  ChevronDown,
     8→  Maximize2,
     9→  Minimize2,
    10→  CheckCircle2,
    11→  Lock,
    12→  Loader2,
    13→  BookOpen,
    14→  Trophy,
    15→  FileText,
    16→  AlertTriangle,
    17→  Rocket,
    18→} from "lucide-react";
    19→import ReactMarkdown from "react-markdown";
    20→import { useAppStore, type CourseData, type CourseChapter, type QuizQuestion } from "@/lib/store";
    21→import { t } from "@/lib/i18n";
    22→import Confetti from "@/components/coursia/Confetti";
    23→import CoursiaLogo from "@/components/coursia/CoursiaLogo";
    24→
    25→const LEVEL_EMOJIS = ["🌱", "⚡", "🔥"];
    26→const LEVEL_NAMES_FR = ["Débutant", "Intermédiaire", "Avancé"];
    27→const LEVEL_NAMES_EN = ["Beginner", "Intermediate", "Advanced"];
    28→
    29→export default function CourseViewer() {
    30→  const lang = useAppStore((s) => s.lang);
    31→  const tx = t(lang);
    32→  const user = useAppStore((s) => s.user);
    33→
    34→  const [course, setCourse] = useState<CourseData | null>(null);
    35→  const [loading, setLoading] = useState(true);
    36→  const [fetchError, setFetchError] = useState(false);
    37→  const [studySessionId, setStudySessionId] = useState<string | null>(null);
    38→  const [showConfetti, setShowConfetti] = useState(false);
    39→  const [courseCompleted, setCourseCompleted] = useState(false);
    40→  const [isCompleting, setIsCompleting] = useState(false);
    41→  const [mobileChapterOpen, setMobileChapterOpen] = useState(false);
    42→
    43→  // Subscription state for free chapter limit
    44→  const [isSubscribed, setIsSubscribed] = useState(false);
    45→  const [freeChapterLimit, setFreeChapterLimit] = useState(1);
    46→
    47→  // Level system states
    48→  const [isGeneratingLevel, setIsGeneratingLevel] = useState(false);
    49→  const [showReviewScreen, setShowReviewScreen] = useState(false);
    50→  const [completedLevel, setCompletedLevel] = useState(-1);
    51→  const [showStopConfirm, setShowStopConfirm] = useState(false);
    52→  const [showAllMastered, setShowAllMastered] = useState(false);
    53→
    54→  const selectedCourseId = useAppStore((s) => s.selectedCourseId);
    55→  const currentChapterIndex = useAppStore((s) => s.currentChapterIndex);
    56→  const setCurrentChapterIndex = useAppStore((s) => s.setCurrentChapterIndex);
    57→  const isFullscreen = useAppStore((s) => s.isFullscreen);
    58→  const setIsFullscreen = useAppStore((s) => s.setIsFullscreen);
    59→  const showFinalQuiz = useAppStore((s) => s.showFinalQuiz);
    60→  const setShowFinalQuiz = useAppStore((s) => s.setShowFinalQuiz);
    61→  const setView = useAppStore((s) => s.setView);
    62→  const showCelebration = useAppStore((s) => s.showCelebration);
    63→  const setShowCelebration = useAppStore((s) => s.setShowCelebration);
    64→  const setCelebrationMessage = useAppStore((s) => s.setCelebrationMessage);
    65→  const setRandomTopic = useAppStore((s) => s.setRandomTopic);
    66→  const setRandomCourseLang = useAppStore((s) => s.setRandomCourseLang);
    67→
    68→  const chapterListRef = useRef<HTMLDivElement>(null);
    69→  const celebrationTimerRef = useRef<ReturnType<typeof setTimeout>>();
    70→  const finalQuizTimerRef = useRef<ReturnType<typeof setTimeout>>();
    71→
    72→  // ── Expanded chapters state for sub-chapter display ──
    73→  const [expandedChapters, setExpandedChapters] = useState<Set<number>>(new Set());
    74→
    75→  const toggleChapterExpanded = (idx: number) => {
    76→    setExpandedChapters((prev) => {
    77→      const next = new Set(prev);
    78→      if (next.has(idx)) next.delete(idx); else next.add(idx);
    79→      return next;
    80→    });
    81→  };
    82→
    83→  // ── Parse sub-chapters from markdown content ──
    84→  const parseSubChapters = useCallback((content: string): string[] => {
    85→    return content.match(/^## (.+)$/gm)?.map((s) => s.replace(/^## /, "")) || [];
    86→  }, []);
    87→
    88→  const chapterSubChapters = useMemo(() => {
    89→    if (!course) return {} as Record<string, string[]>;
    90→    const map: Record<string, string[]> = {};
    91→    for (const ch of course.chapters) map[ch.id] = parseSubChapters(ch.content || "");
    92→    return map;
    93→  }, [course, parseSubChapters]);
    94→
    95→  // ── Group chapters by level ──
    96→  const chaptersByLevel = useMemo(() => {
    97→    if (!course) return [];
    98→    const levels: { level: number; chapters: { chapter: CourseChapter; globalIdx: number }[] }[] = [];
    99→    let currentLevel = -1;
   100→    for (let i = 0; i < course.chapters.length; i++) {
   101→      const ch = course.chapters[i];
   102→      const chLevel = ch.level ?? 0;
   103→      if (chLevel !== currentLevel) {
   104→        levels.push({ level: chLevel, chapters: [] });
   105→        currentLevel = chLevel;
   106→      }
   107→      levels[levels.length - 1].chapters.push({ chapter: ch, globalIdx: i });
   108→    }
   109→    return levels;
   110→  }, [course]);
   111→
   112→  // ── Get current max level and stopped state ──
   113→  const maxUnlockedLevel = course?.maxUnlockedLevel ?? 0;
   114→  const stoppedAtLevel = course?.stoppedAtLevel ?? -1;
   115→  const isStopped = stoppedAtLevel >= 0;
   116→
   117→  // ── Completed count and progress ──
   118→  const completedCount = useMemo(() => {
   119→    if (!course) return 0;
   120→    return course.chapters.filter((ch) => ch.progress?.completed).length;
   121→  }, [course]);
   122→
   123→  const totalChapters = course?.chapters.length ?? 0;
   124→  const overallProgress = totalChapters > 0 ? Math.round((completedCount / totalChapters) * 100) : 0;
   125→
   126→  // Check if all chapters in current max level are completed
   127→  const allCurrentLevelCompleted = useMemo(() => {
   128→    if (!course) return false;
   129→    const levelChapters = course.chapters.filter((ch) => (ch.level ?? 0) <= maxUnlockedLevel);
   130→    if (levelChapters.length === 0) return false;
   131→    return levelChapters.every((ch) => ch.progress?.completed);
   132→  }, [course, maxUnlockedLevel]);
   133→
   134→  const allChaptersCompleted = totalChapters > 0 && completedCount === totalChapters;
   135→
   136→  // ── Check if a chapter is level-locked ──
   137→  const isChapterLevelLocked = (index: number): boolean => {
   138→    if (!course) return false;
   139→    const ch = course.chapters[index];
   140→    const chLevel = ch.level ?? 0;
   141→    // Locked if chapter level exceeds maxUnlockedLevel
   142→    if (chLevel > maxUnlockedLevel) return true;
   143→    // If stopped, lock chapters above stoppedAtLevel
   144→    if (isStopped && chLevel > stoppedAtLevel) return true;
   145→    return false;
   146→  };
   147→
   148→  const isChapterUnlocked = (index: number) => {
   149→    if (isChapterLevelLocked(index)) return false;
   150→    // Free preview: lock chapters beyond freeChapterLimit for non-subscribers
   151→    if (!isSubscribed && index >= freeChapterLimit) return false;
   152→    if (index === 0) return true;
   153→    return course?.chapters[index - 1]?.progress?.completed === true;
   154→  };
   155→
   156→  // ── Study session tracking ──
   157→  const startStudySession = useCallback(async (cId: string, chId?: string) => {
   158→    try {
   159→      const res = await fetch("/api/study-time", {
   160→        method: "POST",
   161→        headers: { "Content-Type": "application/json" },
   162→        body: JSON.stringify({ action: "start", courseId: cId, chapterId: chId }),
   163→      });
   164→      if (res.ok) {
   165→        const data = await res.json();
   166→        setStudySessionId(data.sessionId);
   167→      }
   168→    } catch { /* ignore */ }
   169→  }, []);
   170→
   171→  const endStudySession = useCallback(async () => {
   172→    if (!studySessionId) return;
   173→    try {
   174→      await fetch("/api/study-time", {
   175→        method: "POST",
   176→        headers: { "Content-Type": "application/json" },
   177→        body: JSON.stringify({ action: "end", sessionId: studySessionId }),
   178→      });
   179→    } catch { /* ignore */ }
   180→    setStudySessionId(null);
   181→  }, [studySessionId]);
   182→
   183→  useEffect(() => {
   184→    return () => {
   185→      if (celebrationTimerRef.current) clearTimeout(celebrationTimerRef.current);
   186→      if (finalQuizTimerRef.current) clearTimeout(finalQuizTimerRef.current);
   187→    };
   188→  }, []);
   189→
   190→  useEffect(() => {
   191→    return () => {
   192→      if (studySessionId) {
   193→        fetch("/api/study-time", {
   194→          method: "POST",
   195→          headers: { "Content-Type": "application/json" },
   196→          body: JSON.stringify({ action: "end", sessionId: studySessionId }),
   197→        }).catch(() => {});
   198→      }
   199→    };
   200→  }, [studySessionId]);
   201→
   202→  useEffect(() => {
   203→    const handleBeforeUnload = () => {
   204→      if (studySessionId) {
   205→        navigator.sendBeacon("/api/study-time", new Blob([JSON.stringify({ action: "end", sessionId: studySessionId })], { type: "application/json" }));
   206→      }
   207→    };
   208→    window.addEventListener("beforeunload", handleBeforeUnload);
   209→    return () => window.removeEventListener("beforeunload", handleBeforeUnload);
   210→  }, [studySessionId]);
   211→
   212→  const fetchCourse = useCallback(async () => {
   213→    if (!selectedCourseId) return;
   214→    setLoading(true);
   215→    setFetchError(false);
   216→    try {
   217→      // Fetch subscription status in parallel with course data
   218→      const userId = useAppStore.getState().userId;
   219→      const authHeaders: Record<string, string> = {};
   220→      if (userId) authHeaders["Authorization"] = `Bearer ${userId}`;
   221→
   222→      const [courseRes, statusRes] = await Promise.all([
   223→        fetch(`/api/courses/${selectedCourseId}`),
   224→        userId ? fetch("/api/courses/paywall-status", { headers: authHeaders }) : Promise.resolve(null),
   225→      ]);
   226→
   227→      const data = await courseRes.json();
   228→      // Parse subscription status
   229→      if (statusRes && statusRes.ok) {
   230→        const pw = await statusRes.json();
   231→        setIsSubscribed(!!pw.hasSubscription && pw.subscriptionStatus === "active");
   232→        setFreeChapterLimit(pw.freeChapterLimit || 1);
   233→      }
   234→
   235→      if (courseRes.ok && data.chapters?.length > 0) {
   236→        setCourse(data);
   237→        if (data.courseCompleted) setCourseCompleted(true);
   238→        const savedChapterKey = `coursia-last-chapter-${selectedCourseId}`;
   239→        const savedChapter = typeof window !== "undefined" ? localStorage.getItem(savedChapterKey) : null;
   240→        let restoreIdx = 0;
   241→        if (savedChapter) {
   242→          const savedIdx = parseInt(savedChapter, 10);
   243→          if (!isNaN(savedIdx) && savedIdx >= 0 && savedIdx < data.chapters.length) {
   244→            if (!isChapterLevelLockedForData(savedIdx, data)) restoreIdx = savedIdx;
   245→          }
   246→        }
   247→        if (restoreIdx === 0) {
   248→          const firstIncomplete = data.chapters.findIndex((ch: CourseChapter) => !ch.progress?.completed && !isChapterLevelLockedForData(data.chapters.indexOf(ch), data));
   249→          if (firstIncomplete >= 0) restoreIdx = firstIncomplete;
   250→        }
   251→        const currentIdx = useAppStore.getState().currentChapterIndex;
   252→        if (currentIdx === 0 || currentIdx >= data.chapters.length) {
   253→          setCurrentChapterIndex(restoreIdx);
   254→        }
   255→        startStudySession(selectedCourseId, data.chapters[restoreIdx >= 0 ? restoreIdx : 0]?.id);
   256→      } else {
   257→        setFetchError(true);
   258→      }
   259→    } catch {
   260→      setFetchError(true);
   261→    } finally {
   262→      setLoading(false);
   263→    }
   264→  }, [selectedCourseId, setCurrentChapterIndex]);
   265→
   266→  // Helper to check level lock for data during fetch (before state is set)
   267→  const isChapterLevelLockedForData = (index: number, data: CourseData): boolean => {
   268→    const ch = data.chapters[index];
   269→    const chLevel = ch.level ?? 0;
   270→    if (chLevel > (data.maxUnlockedLevel ?? 0)) return true;
   271→    if ((data.stoppedAtLevel ?? -1) >= 0 && chLevel > (data.stoppedAtLevel ?? -1)) return true;
   272→    return false;
   273→  };
   274→
   275→  useEffect(() => { fetchCourse(); }, [selectedCourseId, fetchCourse]);
   276→
   277→  useEffect(() => {
   278→    if (!chapterListRef.current) return;
   279→    const activeEl = chapterListRef.current.querySelector(`[data-chapter-idx="${currentChapterIndex}"]`);
   280→    if (activeEl) activeEl.scrollIntoView({ behavior: "smooth", block: "nearest" });
   281→  }, [currentChapterIndex]);
   282→
   283→  useEffect(() => {
   284→    if (!course) return;
   285→    const activeCh = course.chapters[currentChapterIndex];
   286→    if (!activeCh) return;
   287→    const subs = chapterSubChapters[activeCh.id];
   288→    if (subs && subs.length > 0) {
   289→      setExpandedChapters((prev) => {
   290→        if (prev.has(currentChapterIndex)) return prev;
   291→        const next = new Set(prev);
   292→        next.add(currentChapterIndex);
   293→        return next;
   294→      });
   295→    }
   296→  }, [currentChapterIndex, course, chapterSubChapters]);
   297→
   298→  useEffect(() => {
   299→    if (!selectedCourseId || !course) return;
   300→    localStorage.setItem(`coursia-last-chapter-${selectedCourseId}`, String(currentChapterIndex));
   301→  }, [currentChapterIndex, selectedCourseId, course]);
   302→
   303→  const currentChapter = course?.chapters[currentChapterIndex];
   304→
   305→  const completeCurrentChapter = useCallback(async (): Promise<boolean> => {
   306→    if (!currentChapter || currentChapter.progress?.completed) return true;
   307→    try {
   308→      const res = await fetch(`/api/courses/${selectedCourseId}/chapters/${currentChapter.id}/complete`, {
   309→        method: "POST",
   310→        headers: { "Content-Type": "application/json" },
   311→      });
   312→      if (res.ok) {
   313→        const courseRes = await fetch(`/api/courses/${selectedCourseId}`);
   314→        if (courseRes.ok) setCourse(await courseRes.json());
   315→        return true;
   316→      }
   317→      return false;
   318→    } catch { return false; }
   319→  }, [currentChapter, selectedCourseId]);
   320→
   321→  const goToNext = useCallback(async () => {
   322→    if (!course || isCompleting) return;
   323→    if (currentChapterIndex >= course.chapters.length - 1) return;
   324→    if (isChapterLevelLocked(currentChapterIndex + 1)) return;
   325→
   326→    // Free preview: block chapter 2+ for non-subscribers
   327→    if (!isSubscribed && currentChapterIndex + 1 >= freeChapterLimit) {
   328→      setView("offers");
   329→      return;
   330→    }
   331→
   332→    setIsCompleting(true);
   333→    const wasJustCompleted = !currentChapter?.progress?.completed;
   334→    if (wasJustCompleted) {
   335→      const success = await completeCurrentChapter();
   336→      if (!success) { setIsCompleting(false); return; }
   337→      const userName = user?.firstName || (lang === "fr" ? "Champion" : "Champion");
   338→      setShowConfetti(true);
   339→      setShowCelebration(true);
   340→      setCelebrationMessage(lang === "fr" ? `Chapitre ${currentChapterIndex + 1} terminé ${userName} ! 🎉` : `Chapter ${currentChapterIndex + 1} done ${userName}! 🎉`);
   341→      if (celebrationTimerRef.current) clearTimeout(celebrationTimerRef.current);
   342→      celebrationTimerRef.current = setTimeout(() => { setShowCelebration(false); setShowConfetti(false); }, 2000);
   343→    }
   344→    endStudySession();
   345→    const nextIdx = currentChapterIndex + 1;
   346→    setCurrentChapterIndex(nextIdx);
   347→    startStudySession(course.id, course.chapters[nextIdx]?.id);
   348→    setIsCompleting(false);
   349→  }, [course, isCompleting, currentChapter?.progress?.completed, currentChapterIndex, completeCurrentChapter, user?.firstName, lang, endStudySession, startStudySession, setCurrentChapterIndex, isSubscribed, freeChapterLimit, setView]);
   350→
   351→  const goToPrev = useCallback(() => {
   352→    if (currentChapterIndex === 0 || !course) return;
   353→    endStudySession();
   354→    const prevIdx = currentChapterIndex - 1;
   355→    setCurrentChapterIndex(prevIdx);
   356→    startStudySession(course.id, course.chapters[prevIdx]?.id);
   357→  }, [currentChapterIndex, course, endStudySession, startStudySession, setCurrentChapterIndex]);
   358→
   359→  // ── Level Complete handler ──
   360→  const handleFinalQuizComplete = useCallback((passed: boolean) => {
   361→    if (!passed) return;
   362→    const userName = user?.firstName || (lang === "fr" ? "Champion" : "Champion");
   363→    setShowConfetti(true);
   364→    setShowCelebration(true);
   365→    setCelebrationMessage(lang === "fr" ? `Félicitations ${userName} ! 🏆` : `Congratulations ${userName}! 🏆`);
   366→    setCourseCompleted(true);
   367→    if (finalQuizTimerRef.current) clearTimeout(finalQuizTimerRef.current);
   368→    finalQuizTimerRef.current = setTimeout(() => {
   369→      setShowCelebration(false);
   370→      setShowFinalQuiz(false);
   371→      setShowConfetti(false);
   372→      endStudySession();
   373→
   374→      const courseLevel = course?.level ?? 0;
   375→      const currentMaxLevel = course?.maxUnlockedLevel ?? 0;
   376→      setCompletedLevel(currentMaxLevel);
   377→
   378→      if (courseLevel >= 2 && currentMaxLevel >= 2) {
   379→        // All levels mastered!
   380→        setShowAllMastered(true);
   381→      } else {
   382→        // Show level-up modal
   383→        setShowReviewScreen(true);
   384→      }
   385→      fetchCourse();
   386→    }, 4000);
   387→  }, [course, user?.firstName, lang, endStudySession, fetchCourse]);
   388→
   389→  // ── Continue to next level ──
   390→  const handleContinueToNextLevel = useCallback(async () => {
   391→    if (!course || !selectedCourseId) return;
   392→    const nextLevel = maxUnlockedLevel + 1;
   393→    if (nextLevel > 2) return;
   394→
   395→    setIsGeneratingLevel(true);
   396→    setShowReviewScreen(false);
   397→    setShowConfetti(false);
   398→
   399→    try {
   400→      const res = await fetch(`/api/courses/${selectedCourseId}/generate-level`, {
   401→        method: "POST",
   402→        headers: { "Content-Type": "application/json" },
   403→        body: JSON.stringify({ level: nextLevel }),
   404→      });
   405→      const data = await res.json();
   406→      if (res.ok && data.chapters?.length > 0) {
   407→        // Refetch course
   408→        await fetchCourse();
   409→        // Navigate to first chapter of new level
   410→        const firstNewChapterIdx = course.chapters.findIndex((ch) => (ch.level ?? 0) === nextLevel);
   411→        if (firstNewChapterIdx >= 0) setCurrentChapterIndex(firstNewChapterIdx);
   412→      }
   413→    } catch {
   414→      // show error
   415→    } finally {
   416→      setIsGeneratingLevel(false);
   417→    }
   418→  }, [course, selectedCourseId, maxUnlockedLevel, fetchCourse, setCurrentChapterIndex]);
   419→
   420→  // ── Stop at current level ──
   421→  const handleStopHere = useCallback(async () => {
   422→    if (!selectedCourseId) return;
   423→    try {
   424→      await fetch(`/api/courses/${selectedCourseId}/stop-level`, {
   425→        method: "POST",
   426→        headers: { "Content-Type": "application/json" },
   427→      });
   428→      setShowReviewScreen(false);
   429→      setShowConfetti(false);
   430→      setShowStopConfirm(true);
   431→      fetchCourse();
   432→    } catch {
   433→      // ignore
   434→    }
   435→  }, [selectedCourseId, fetchCourse]);
   436→
   437→  const handleCloseStopConfirm = () => {
   438→    setShowStopConfirm(false);
   439→  };
   440→
   441→  const handleCloseAllMastered = () => {
   442→    setShowAllMastered(false);
   443→    setView("library");
   444→  };
   445→
   446→  // Keyboard navigation
   447→  useEffect(() => {
   448→    const handleKey = (e: KeyboardEvent) => {
   449→      if (isFullscreen) {
   450→        if (e.key === "ArrowRight" || e.key === " ") { e.preventDefault(); goToNext(); }
   451→        if (e.key === "ArrowLeft") { e.preventDefault(); goToPrev(); }
   452→        if (e.key === "Escape") setIsFullscreen(false);
   453→      }
   454→    };
   455→    window.addEventListener("keydown", handleKey);
   456→    return () => window.removeEventListener("keydown", handleKey);
   457→  }, [isFullscreen, goToNext, goToPrev, setIsFullscreen]);
   458→
   459→  // ── Level names helper ──
   460→  const getLevelName = (level: number) => lang === "fr" ? LEVEL_NAMES_FR[level] : LEVEL_NAMES_EN[level];
   461→
   462→  // ══════════════════════════════════════════════════════════════════
   463→  // Loading / error state
   464→  // ══════════════════════════════════════════════════════════════════
   465→  if (!selectedCourseId || fetchError || (!loading && !course)) {
   466→    return (
   467→      <div className="flex items-center justify-center min-h-screen animate-fade-in">
   468→        <div className="text-center">
   469→          <div className="text-5xl mb-4 opacity-40">📚</div>
   470→          <p className="text-muted-foreground text-lg mb-6">{tx.viewer.back}</p>
   471→          <button onClick={() => setView("landing")} className="px-6 py-3 rounded-full bg-gradient-to-r from-mauve to-mauve-dark text-white font-bold hover:from-mauve-light hover:to-mauve transition-all cursor-pointer">{tx.viewer.back}</button>
   472→        </div>
   473→      </div>
   474→    );
   475→  }
   476→
   477→  if (loading || !course || !currentChapter) {
   478→    return (
   479→      <div className="flex items-center justify-center min-h-screen animate-fade-in">
   480→        <div className="text-center">
   481→          <Loader2 className="w-10 h-10 animate-spin text-mauve mx-auto mb-4" />
   482→          <p className="text-muted-foreground text-lg">{tx.viewer.loading}</p>
   483→        </div>
   484→      </div>
   485→    );
   486→  }
   487→
   488→  // ══════════════════════════════════════════════════════════════════
   489→  // GENERATING NEXT LEVEL — full screen loading
   490→  // ══════════════════════════════════════════════════════════════════
   491→  if (isGeneratingLevel) {
   492→    return (
   493→      <>
   494→        <Confetti active={true} />
   495→        <div className="fixed inset-0 z-50 bg-night flex flex-col items-center justify-center animate-fade-in">
   496→          <div className="text-7xl mb-6 animate-bounce">🚀</div>
   497→          <h2 className="text-2xl font-extrabold gradient-text mb-3">
   498→            {lang === "fr" ? "Génération du prochain niveau..." : "Generating next level..."}
   499→          </h2>
   500→          <p className="text-muted-foreground text-sm mb-6">
   501→            {lang === "fr"
   502→              ? `Niveau ${getLevelName(maxUnlockedLevel + 1)} en cours de création`

  1086→  isFinalQuiz: boolean;
  1087→  onComplete: (passed: boolean) => void;
  1088→  onBack: () => void;
  1089→}) {
  1090→  const lang = useAppStore((s) => s.lang);
  1091→  const tx = t(lang);
  1092→
  1093→  const [questions, setQuestions] = useState<QuizQuestion[]>([]);
  1094→  const [answers, setAnswers] = useState<Record<number, number>>({});
  1095→  const [loading, setLoading] = useState(true);
  1096→  const [submitting, setSubmitting] = useState(false);

  1098→

  1100→    const fetchQuiz = async () => {
  1101→      try {
  1102→        const url = `/api/courses/${courseId}/final-quiz`;

  1104→          method: "POST",

  1107→        });
  1108→        const data = await res.json();

  1110→          setQuestions(data.quiz.questions);
  1111→        }
  1112→      } catch { /* fail silently */ }

  1117→
  1118→  const submitQuiz = async () => {

  1120→    try {
  1121→      const url = `/api/courses/${courseId}/final-quiz`;
  1122→      const res = await fetch(url, {
  1123→        method: "PUT",
  1124→        headers: { "Content-Type": "application/json" },
  1125→        body: JSON.stringify({ answers }),
  1126→      });
  1127→      const data = await res.json();
  1128→      if (res.ok && data.result) {
  1129→        setResult(data.result);
  1130→        if (data.result.passed) {

  1140→      <div className="flex items-center justify-center py-12">

  1204→        >
  1205→          {submitting ? <Loader2 className="w-4 h-4 animate-spin" /> : tx.viewer.submit}
